#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="sentiment-history.html"

python - <<'PY'
from pathlib import Path

p = Path("sentiment-history.html")
text = p.read_text(encoding="utf-8")

# Add one shared momentum formula immediately before drawBars.
anchor = '    function drawBars(canvas, points, colors=[COLORS.gold,COLORS.pink,COLORS.purple]){\n'
if anchor not in text:
    raise SystemExit("STOP: drawBars() anchor not found.")

if "function sentimentMomentumIndex(" not in text:
    fn = '''    function sentimentMomentumIndex(current, previous){
      if(!Number.isFinite(current) || !Number.isFinite(previous)) return 50;
      return clamp(50 + (current - previous) * 3, 0, 100);
    }

'''
    text = text.replace(anchor, fn + anchor, 1)

old_start = '''    function drawBars(canvas, points, colors=[COLORS.gold,COLORS.pink,COLORS.purple]){
      const {ctx,w,h}=setupCanvas(canvas);
      const box=drawGrid(ctx,w,h,{l:42,r:14,t:34,b:26});
      const pts=points.slice(-5);
      if(!pts.length) return;
'''

new_start = '''    function drawBars(canvas, points, colors=[COLORS.gold,COLORS.pink,COLORS.purple]){
      const {ctx,w,h}=setupCanvas(canvas);
      const box=drawGrid(ctx,w,h,{l:42,r:14,t:34,b:26});
      const withMomentum=points.map((p,i,a)=>({
        ...p,
        momentum:i
          ? sentimentMomentumIndex(p.global,a[i-1].global)
          : 50
      }));
      const pts=withMomentum.slice(-5);
      if(!pts.length) return;
'''

if old_start not in text:
    raise SystemExit("STOP: expected drawBars() opening block not found.")
text = text.replace(old_start, new_start, 1)

old_fake = '''        const momentum = Math.abs((p.us ?? p.value) - (p.global ?? p.value)) + 35;
'''
new_real = '''        const momentum = p.momentum;
'''
if old_fake not in text:
    raise SystemExit("STOP: old Chart 3 momentum formula not found.")
text = text.replace(old_fake, new_real, 1)

old_mom = '''      const mom=momBase.map((p,i,a)=>({label:p.label,value:i?clamp(50+(p.value-a[i-1].value)*3,0,100):50}));
'''
new_mom = '''      const mom=momBase.map((p,i,a)=>({label:p.label,value:i?sentimentMomentumIndex(p.value,a[i-1].value):50}));
'''
if old_mom not in text:
    raise SystemExit("STOP: existing momentum-chart formula not found.")
text = text.replace(old_mom, new_mom, 1)

p.write_text(text, encoding="utf-8")
print("Chart 3 momentum formula standardized.")
PY

git diff --check

echo
echo "Changed file only:"
git status --short
echo
git diff --stat -- "$FILE"
echo
echo "Momentum markers:"
grep -n -E 'sentimentMomentumIndex|const momentum = p\.momentum|const mom=momBase' "$FILE"
echo
echo "No commit or push was performed."
