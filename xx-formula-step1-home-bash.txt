#!/usr/bin/env bash
set -euo pipefail

# PublicSentimentDash Formula Centralization
# STEP 1 — HOME PAGE ONLY
#
# This changes exactly one production page: index.html
# It does NOT modify sentiment-core.js, engine files, CSS, ribbon, voting,
# dashboard, history, Market Pulse, or any other page.
#
# Goal:
# Home Global/U.S./Europe gauges keep backend scores + confidence,
# but their displayed sentiment LABEL always comes from the centralized
# PSDCore.classifySentiment() scale already used by scoreLabel().
#
# Expected example after this change:
# 47 -> Mixed/Neutral
# 52 -> Mixed/Neutral
# 56 -> Bullish

FILE="index.html"

if [ ! -f "$FILE" ]; then
  echo "ERROR: index.html not found. Run this from the repository root."
  exit 1
fi

python3 - "$FILE" <<'PY'
from pathlib import Path
import sys

path = Path(sys.argv[1])
text = path.read_text(encoding="utf-8")

old_lines = [
    'updatePulseCard("Global", global.score, global.label || scoreLabel(global.score), pulseCountText("Global", global));',
    'updatePulseCard("US", us.score, us.label || scoreLabel(us.score), pulseCountText("U.S.", us));',
    'updatePulseCard("Europe", europe.score, europe.label || scoreLabel(europe.score), pulseCountText("Europe", europe));',
]

new_lines = [
    'updatePulseCard("Global", global.score, scoreLabel(global.score), pulseCountText("Global", global));',
    'updatePulseCard("US", us.score, scoreLabel(us.score), pulseCountText("U.S.", us));',
    'updatePulseCard("Europe", europe.score, scoreLabel(europe.score), pulseCountText("Europe", europe));',
]

# Safety: require the centralized classifier hook to already exist.
if "window.PSDCore.classifySentiment" not in text:
    raise SystemExit("ERROR: Central scoreLabel()/PSDCore classifier was not found. No change made.")

# Safety: require exactly one current render line for each gauge.
for old in old_lines:
    count = text.count(old)
    if count != 1:
        raise SystemExit(
            f"ERROR: Expected exactly one matching Home gauge line, found {count}. "
            "No change made."
        )

original = text
for old, new in zip(old_lines, new_lines):
    text = text.replace(old, new, 1)

# Post-change validation.
for old in old_lines:
    if old in text:
        raise SystemExit("ERROR: Old backend-label rendering still remains. No file written.")

for new in new_lines:
    if text.count(new) != 1:
        raise SystemExit("ERROR: New centralized Home gauge rendering validation failed.")

# Confirm we did not alter unrelated shared-file references.
required = [
    'sentiment-core.js',
    'site-nav.js',
    'site-theme.js',
]
for token in required:
    if token not in text:
        raise SystemExit(f"ERROR: Existing shared dependency {token} is missing.")

path.write_text(text, encoding="utf-8")

print("STEP 1 COMPLETE — Home only")
print("Changed file: index.html")
print("")
print("Home gauge labels now use centralized scoreLabel()/PSDCore.classifySentiment().")
print("Backend regional scores and confidence are unchanged.")
print("No other page or shared file was modified by this script.")
PY

echo
echo "Run this next to inspect the exact change:"
echo "  git diff -- index.html"
echo
echo "Do NOT commit other files with this step."
