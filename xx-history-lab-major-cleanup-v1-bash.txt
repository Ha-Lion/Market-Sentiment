js = r'''
  <script id="xx-history-clean-lab-js">
  (() => {
    const REGION_META = {
      us:{label:"U.S.A.", css:"--hist-pink", fallback:"#ff3f86"},
      global:{label:"Global", css:"--hist-gold", fallback:"#f6c85f"},
      europe:{label:"Europe", css:"--hist-cyan", fallback:"#4adeff"}
    };

    const selected = new Set(["us","global"]);
    let historyData = null;
    let dashboardData = null;
    let resizeTimer = null;

    const finite = v => Number.isFinite(Number(v));
    const num = v => finite(v) ? Number(v) : null;

    function regionColor(key){
      const meta=REGION_META[key];
      const css=getComputedStyle(document.documentElement).getPropertyValue(meta.css).trim();
      return css || meta.fallback;
    }

    function avg(values){
      const vals=values.filter(Number.isFinite);
      return vals.length ? vals.reduce((a,b)=>a+b,0)/vals.length : null;
    }

    function dateLabel(d){
      return new Intl.DateTimeFormat("en-US",{month:"short",day:"numeric",timeZone:"America/New_York"}).format(d);
    }

    function timeLabel(d){
      return new Intl.DateTimeFormat("en-US",{hour:"numeric",minute:"2-digit",hour12:true,timeZone:"America/New_York"})
        .format(d).replace(" ","");
    }

    function monthLabel(key){
      const [y,m]=key.split("-").map(Number);
      return new Intl.DateTimeFormat("en-US",{month:"short",year:"2-digit"}).format(new Date(y,m-1,1));
    }

    function isoWeekParts(date){
      const d=new Date(Date.UTC(date.getUTCFullYear(),date.getUTCMonth(),date.getUTCDate()));
      const day=d.getUTCDay() || 7;
      d.setUTCDate(d.getUTCDate()+4-day);
      const yearStart=new Date(Date.UTC(d.getUTCFullYear(),0,1));
      const week=Math.ceil((((d-yearStart)/86400000)+1)/7);
      return {year:d.getUTCFullYear(),week};
    }

    function normalize(){
      const daily=(historyData?.records || []).map(r => {
        const ts=new Date(String(r.date)+"T12:00:00-04:00");
        return {
          ts,key:String(r.date),label:dateLabel(ts),
          global:num(r.global_psi),us:num(r.us_psi),
          europe:num(r.europe_psi ?? r.euro_psi)
        };
      }).filter(p => !Number.isNaN(p.ts.getTime())).sort((a,b)=>a.ts-b.ts);

      const hourly=(historyData?.intraday || []).map(r => {
        const ts=new Date(r.ts || r.sample_key);
        return {
          ts,key:String(r.ts || r.sample_key || ""),label:timeLabel(ts),
          global:num(r.global_psi),us:num(r.us_psi),
          europe:num(r.europe_psi ?? r.euro_psi)
        };
      }).filter(p => !Number.isNaN(p.ts.getTime())).sort((a,b)=>a.ts-b.ts).slice(-12);

      const weeklyMap=new Map();
      daily.forEach(p=>{
        const {year,week}=isoWeekParts(p.ts);
        const key=`${year}-W${String(week).padStart(2,"0")}`;
        if(!weeklyMap.has(key)){
          weeklyMap.set(key,{key,label:`W${week}`,ts:p.ts,global:[],us:[],europe:[],count:0});
        }
        const g=weeklyMap.get(key);
        ["global","us","europe"].forEach(k=>{ if(Number.isFinite(p[k])) g[k].push(p[k]); });
        g.count++;
        if(p.ts>g.ts) g.ts=p.ts;
      });
      const weekly=[...weeklyMap.values()]
        .map(g=>({
          key:g.key,label:g.label,ts:g.ts,count:g.count,
          global:avg(g.global),us:avg(g.us),europe:avg(g.europe)
        }))
        .sort((a,b)=>a.ts-b.ts).slice(-8);

      const monthlyMap=new Map();
      daily.forEach(p=>{
        const y=p.ts.getFullYear();
        const m=String(p.ts.getMonth()+1).padStart(2,"0");
        const key=`${y}-${m}`;
        if(!monthlyMap.has(key)){
          monthlyMap.set(key,{key,label:monthLabel(key),ts:p.ts,global:[],us:[],europe:[],count:0});
        }
        const g=monthlyMap.get(key);
        ["global","us","europe"].forEach(k=>{ if(Number.isFinite(p[k])) g[k].push(p[k]); });
        g.count++;
        if(p.ts>g.ts) g.ts=p.ts;
      });
      const monthly=[...monthlyMap.values()]
        .map(g=>({
          key:g.key,label:g.label,ts:g.ts,count:g.count,
          global:avg(g.global),us:avg(g.us),europe:avg(g.europe)
        }))
        .sort((a,b)=>a.ts-b.ts).slice(-6);

      return {hourly,daily:daily.slice(-14),weekly,monthly};
    }

    function setupCanvas(canvas){
      const rect=canvas.getBoundingClientRect();
      const ratio=window.devicePixelRatio || 1;
      canvas.width=Math.max(320,Math.floor(rect.width*ratio));
      canvas.height=Math.max(180,Math.floor(rect.height*ratio));
      const ctx=canvas.getContext("2d");
      ctx.setTransform(ratio,0,0,ratio,0,0);
      return {ctx,w:rect.width,h:rect.height};
    }

    function renderLegend(period,data){
      const el=document.querySelector(`[data-legend-for="${period}"]`);
      if(!el) return;
      el.innerHTML=[...selected].map(k=>{
        const has=data.some(p=>Number.isFinite(p[k]));
        return `<span class="xx-legend-item" title="${REGION_META[k].label}${has ? "" : " — awaiting history"}">
          <i class="xx-legend-line" style="background:${regionColor(k)}"></i>${REGION_META[k].label}${has ? "" : " *"}
        </span>`;
      }).join("");
    }

    function drawChart(canvasId,emptyId,data,period){
      const canvas=document.getElementById(canvasId);
      const empty=document.getElementById(emptyId);
      if(!canvas || !empty) return;

      renderLegend(period,data);
      const available=[...selected].filter(k=>data.some(p=>Number.isFinite(p[k])));
      const missing=[...selected].filter(k=>!data.some(p=>Number.isFinite(p[k])));

      if(!data.length || !available.length){
        const {ctx,w,h}=setupCanvas(canvas);
        ctx.clearRect(0,0,w,h);
        empty.textContent = missing.includes("europe")
          ? "Europe historical PSI is not stored yet. The Europe line will appear automatically after europe_psi is added to sentiment_history.json."
          : "No historical PSI is available for the selected region.";
        empty.classList.add("show");
        return;
      }

      empty.classList.remove("show");
      empty.removeAttribute("style");

      const {ctx,w,h}=setupCanvas(canvas);
      ctx.clearRect(0,0,w,h);

      const pad={l:34,r:20,t:15,b:28};
      const iw=Math.max(1,w-pad.l-pad.r);
      const ih=Math.max(1,h-pad.t-pad.b);
      const y=v=>pad.t+(100-v)/100*ih;
      const x=i=>data.length===1 ? pad.l+iw/2 : pad.l+(i/(data.length-1))*iw;

      const textColor=getComputedStyle(document.documentElement).getPropertyValue("--hist-muted").trim() || "#94a3b8";
      const gridColor="rgba(148,163,184,.15)";
      const neutralColor="rgba(148,163,184,.34)";

      ctx.font="9px Inter,Arial";
      ctx.lineWidth=1;

      [0,25,50,75,100].forEach(v=>{
        const yy=y(v);
        ctx.strokeStyle=v===50 ? neutralColor : gridColor;
        ctx.beginPath();ctx.moveTo(pad.l,yy);ctx.lineTo(w-pad.r,yy);ctx.stroke();
        ctx.fillStyle=textColor;ctx.textAlign="right";ctx.textBaseline="middle";
        ctx.fillText(String(v),pad.l-5,yy);
      });

      const maxLabels=period==="daily" ? 7 : 8;
      const labelCount=Math.min(maxLabels,data.length);
      const indices=new Set();
      if(labelCount===1) indices.add(0);
      else{
        for(let i=0;i<labelCount;i++) indices.add(Math.round(i*(data.length-1)/(labelCount-1)));
      }

      [...indices].forEach(i=>{
        ctx.fillStyle=textColor;
        ctx.textAlign=i===0 ? "left" : (i===data.length-1 ? "right" : "center");
        ctx.textBaseline="alphabetic";
        ctx.fillText(data[i].label,x(i),h-7);
      });

      available.forEach(key=>{
        const points=data.map((p,i)=>({i,value:p[key]})).filter(p=>Number.isFinite(p.value));
        if(!points.length) return;

        const color=regionColor(key);
        ctx.strokeStyle=color;ctx.lineWidth=2.4;ctx.lineJoin="round";ctx.lineCap="round";
        ctx.beginPath();
        points.forEach((p,j)=>{
          const xx=x(p.i),yy=y(p.value);
          if(j===0) ctx.moveTo(xx,yy); else ctx.lineTo(xx,yy);
        });
        ctx.stroke();

        points.forEach(p=>{
          const xx=x(p.i),yy=y(p.value);
          ctx.fillStyle=color;ctx.beginPath();ctx.arc(xx,yy,3,0,Math.PI*2);ctx.fill();
        });

        const last=points[points.length-1];
        ctx.fillStyle=color;ctx.font="700 9px Inter,Arial";ctx.textAlign="right";ctx.textBaseline="bottom";
        ctx.fillText(`${REGION_META[key].label} ${Math.round(last.value)}`,w-pad.r,Math.max(11,y(last.value)-5));
      });

      if(missing.length){
        empty.textContent=missing.map(k=>REGION_META[k].label).join(" + ")+" selected, but historical data is not stored yet.";
        empty.classList.add("show");
        empty.style.alignItems="end";
        empty.style.justifyItems="center";
        empty.style.paddingBottom="30px";
      }
    }

    function updateButtons(){
      document.querySelectorAll(".xx-region-btn[data-region]").forEach(btn=>{
        const key=btn.dataset.region;
        const active=key==="all"
          ? ["us","global","europe"].every(k=>selected.has(k))
          : selected.has(key);
        btn.classList.toggle("active",active);
        btn.setAttribute("aria-pressed",String(active));
      });
    }

    function render(){
      if(!historyData) return;
      const sets=normalize();
      drawChart("xxChartHourly","xxEmptyHourly",sets.hourly,"hourly");
      drawChart("xxChartDaily","xxEmptyDaily",sets.daily,"daily");
      drawChart("xxChartWeekly","xxEmptyWeekly",sets.weekly,"weekly");
      drawChart("xxChartMonthly","xxEmptyMonthly",sets.monthly,"monthly");

      const hasEurope=[...sets.hourly,...sets.daily].some(p=>Number.isFinite(p.europe));
      const currentEurope=num(dashboardData?.regional_pulse?.europe?.score);
      const status=document.getElementById("xxHistoryStatus");
      if(status){
        if(hasEurope) status.textContent="Global, U.S.A. and Europe historical PSI available.";
        else if(Number.isFinite(currentEurope)) status.textContent=`Europe current PSI ${Math.round(currentEurope)} • historical Europe tracking not stored yet`;
        else status.textContent="Europe historical tracking not stored yet.";
      }
    }

    function bindControls(){
      document.querySelectorAll(".xx-region-btn[data-region]").forEach(btn=>{
        btn.addEventListener("click",()=>{
          const key=btn.dataset.region;
          if(key==="all"){
            selected.clear();
            ["us","global","europe"].forEach(k=>selected.add(k));
          }else if(selected.has(key)){
            if(selected.size>1) selected.delete(key);
          }else{
            selected.add(key);
          }
          updateButtons();
          render();
        });
      });
    }

    async function load(){
      try{
        const stamp=Date.now();
        const [historyRes,dashboardRes]=await Promise.all([
          fetch(`sentiment_history.json?v=${stamp}`,{cache:"no-store"}),
          fetch(`dashboard_data.json?v=${stamp}`,{cache:"no-store"})
        ]);

        if(!historyRes.ok) throw new Error(`sentiment_history.json ${historyRes.status}`);
        historyData=await historyRes.json();
        if(dashboardRes.ok) dashboardData=await dashboardRes.json();

        bindControls();
        updateButtons();
        render();
      }catch(err){
        console.error("Historical lab load failed:",err);
        const status=document.getElementById("xxHistoryStatus");
        if(status) status.textContent="Unable to load historical PSI data.";
      }
    }

    window.addEventListener("resize",()=>{
      clearTimeout(resizeTimer);
      resizeTimer=setTimeout(render,120);
    });

    if(document.readyState==="loading") document.addEventListener("DOMContentLoaded",load,{once:true});
    else load();
  })();
  </script>
'''
