$ErrorActionPreference = "Stop" $Repo = (Get-Location).Path $Required = @( "site-nav.js", "site-preferences.js", "account.html", "my-account.html" ) foreach ($File in $Required) { if (-not (Test-Path (Join-Path $Repo $File))) { throw "Run this from the root of the Market-Sentiment repo. Missing: $File" } } $Package = Split-Path -Parent $MyInvocation.MyCommand.Path # Replace only the two already-created pages and My Account JS. Copy-Item (Join-Path $Package "account.html") (Join-Path $Repo "account.html") -Force Copy-Item (Join-Path $Package "my-account.html") (Join-Path $Repo "my-account.html") -Force Copy-Item (Join-Path $Package "my-account.js") (Join-Path $Repo "my-account.js") -Force function Fix-AccountMenu([string]$Path) { $Text = Get-Content $Path -Raw # Desktop dropdown links $Text = $Text.Replace( 'Preferences', 'Preferences' ) $Text = $Text.Replace( 'Update Information', 'My Account' ) # Mobile/private link filtering references $Text = $Text.Replace( '"account.html#preferences",', '"account.html",' ) $Text = $Text.Replace( '"account.html#profile",', '"my-account.html",' ) Set-Content $Path $Text -Encoding UTF8 } Fix-AccountMenu (Join-Path $Repo "site-nav.js") Fix-AccountMenu (Join-Path $Repo "site-preferences.js") # Verify $Nav = Get-Content (Join-Path $Repo "site-nav.js") -Raw $Prefs = Get-Content (Join-Path $Repo "site-preferences.js") -Raw $Account = Get-Content (Join-Path $Repo "account.html") -Raw $MyAccount = Get-Content (Join-Path $Repo "my-account.html") -Raw if ($Nav -match 'Update Information') { throw "site-nav.js still contains Update Information." } if ($Prefs -match 'Update Information') { throw "site-preferences.js still contains Update Information." } if ($Nav -notmatch 'href="my-account\.html">My Account') { throw "site-nav.js is missing My Account." } if ($Prefs -notmatch 'href="my-account\.html">My Account') { throw "site-preferences.js is missing My Account." } if ($Account -match 'href="my-account\.html"') { throw "Preferences page still contains an internal My Account link." } if ($MyAccount -match 'href="account\.html"') { throw "My Account page still contains an internal Preferences link." } Write-Host "" Write-Host "SUCCESS" Write-Host "Preferences and My Account are now separate pages."
