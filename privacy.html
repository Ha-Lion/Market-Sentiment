$ErrorActionPreference = "Stop" $Repo = (Get-Location).Path Write-Host "===== PUBLIC SENTIMENT DASH: ACCOUNT SPLIT =====" Write-Host "Repository: $Repo" Write-Host "" $Required = @("site-nav.js", "site-preferences.js", "account.html", "privacy.html") foreach ($File in $Required) { if (-not (Test-Path (Join-Path $Repo $File))) { throw "Run this script from the root of the Market-Sentiment repository. Missing: $File" } } $Package = Split-Path -Parent $MyInvocation.MyCommand.Path Copy-Item (Join-Path $Package "account.html") (Join-Path $Repo "account.html") -Force Copy-Item (Join-Path $Package "my-account.html") (Join-Path $Repo "my-account.html") -Force Copy-Item (Join-Path $Package "my-account.js") (Join-Path $Repo "my-account.js") -Force function Patch-MenuFile([string]$Path) { $Text = Get-Content $Path -Raw $Text = $Text.Replace( 'Preferences', 'Preferences' ) $Text = $Text.Replace( 'Update Information', 'My Account' ) $Text = $Text.Replace( '"account.html#preferences",', '"account.html",' ) $Text = $Text.Replace( '"account.html#profile",', '"my-account.html",' ) Set-Content $Path $Text -Encoding UTF8 } Patch-MenuFile (Join-Path $Repo "site-nav.js") Patch-MenuFile (Join-Path $Repo "site-preferences.js") $PrivacyPath = Join-Path $Repo "privacy.html" $Privacy = Get-Content $PrivacyPath -Raw $Old = '
  • Account information: If you create an account, we collect your email address, encrypted authentication credentials, optional display name, account identifiers, confirmation and recovery status, and account creation or sign-in timestamps. Passwords are handled by the authentication provider and are not visible to Public Sentiment Dash in readable form.
    • ' $New = '
  • Account information: If you create an account, we collect your primary email address, encrypted authentication credentials, optional first and last name, optional second email address, optional phone number, optional profile picture/avatar, username or display name, account identifiers, confirmation and recovery status, and account creation or sign-in timestamps. Passwords are handled by the authentication provider and are not visible to Public Sentiment Dash in readable form.
    • ' if ($Privacy.Contains($Old)) { $Privacy = $Privacy.Replace($Old, $New) Set-Content $PrivacyPath $Privacy -Encoding UTF8 } else { Write-Warning "Privacy account-information paragraph was not an exact match. privacy.html was left unchanged." } Write-Host "" Write-Host "===== VERIFY =====" $Nav = Get-Content (Join-Path $Repo "site-nav.js") -Raw $Prefs = Get-Content (Join-Path $Repo "site-preferences.js") -Raw if ($Nav -notmatch 'href="my-account\.html">My Account') { throw "site-nav.js My Account link was not installed." } if ($Prefs -notmatch 'href="my-account\.html">My Account') { throw "site-preferences.js My Account link was not installed." } if ($Nav -match 'Update Information') { throw "site-nav.js still contains Update Information." } if ($Prefs -match 'Update Information') { throw "site-preferences.js still contains Update Information." } Write-Host "OK: account.html is now Preferences" Write-Host "OK: my-account.html created" Write-Host "OK: my-account.js created" Write-Host "OK: dropdown renamed to My Account" Write-Host "OK: mobile signed-out private-link filtering updated" Write-Host "" Write-Host "Review changes with:" Write-Host " git diff -- account.html my-account.html my-account.js site-nav.js site-preferences.js privacy.html" Write-Host "" Write-Host "If the diff looks good:" Write-Host ' git add account.html my-account.html my-account.js site-nav.js site-preferences.js privacy.html' Write-Host ' git commit -m "Split My Account from Preferences"' Write-Host ' git push'
