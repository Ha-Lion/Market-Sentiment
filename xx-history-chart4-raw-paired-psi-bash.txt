#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="sentiment-history.html"

python - <<'PY'
from pathlib import Path

p = Path("sentiment-history.html")
text = p.read_text(encoding="utf-8")

anchor = '''    function updateSkylineHeader(){
'''
if anchor not in text:
    raise SystemExit("STOP: updateSkylineHeader() anchor not found.")

if "function skylineComparisonPoints(" not in text:
    fn = '''    function skylineComparisonPoints(){
      if(state.period === "hourly"){
        return aggregateHourlyComparison(state.hourly)
          .filter(p => Number.isFinite(p.global) && Number.isFinite(p.us))
          .slice(-skylineLimit(state.period));
      }

      if(state.period === "weekly"){
        return aggregateWeeklyComparison(filterByRange(state.raw))
          .filter(p => Number.isFinite(p.global) && Number.isFinite(p.us))
          .slice(-skylineLimit(state.period));
      }

      const records = filterByRange(state.raw);
      const groups = new Map();

      records
        .filter(r =>
          r &&
          r.ts instanceof Date &&
          !Number.isNaN(r.ts.getTime()) &&
          Number.isFinite(r.global) &&
          Number.isFinite(r.us)
        )
        .forEach(r => {
          let key, label, sort;

          if(state.period === "monthly"){
            key = r.date.slice(0,7);
            label = labelMonth(key);
            sort = key;
          }else{
            key = r.date;
            label = labelDate(r.ts);
            sort = r.date;
          }

          if(!groups.has(key)){
            groups.set(key,{
              key,
              label,
              sort,
              global:[],
              us:[],
              count:0,
              lastTs:r.ts
            });
          }

          const g = groups.get(key);
          g.global.push(r.global);
          g.us.push(r.us);
          g.count++;
          if(r.ts > g.lastTs) g.lastTs = r.ts;
        });

      return [...groups.values()]
        .map(g => ({
          key:g.key,
          label:g.label,
          sort:g.sort,
          count:g.count,
          lastTs:g.lastTs,
          global:avg(g.global),
          us:avg(g.us)
        }))
        .sort((a,b) =>
          String(a.sort).localeCompare(String(b.sort)) ||
          a.lastTs-b.lastTs
        )
        .slice(-skylineLimit(state.period));
    }

'''
    text = text.replace(anchor, fn + anchor, 1)

old_header = '''      if(title) title.textContent = label + " Global vs U.S. skyline";
'''
new_header = '''      if(title) title.textContent = label + " Global vs. U.S. PSI";
'''
if old_header not in text:
    raise SystemExit("STOP: expected skyline title line not found.")
text = text.replace(old_header, new_header, 1)

old_draw = '''      updateSkylineHeader();
      const skylineGlobal=movingAverage(skylinePoints("global")).map(p=>({...p,value:p.smooth??p.value}));
      const skylineUs=movingAverage(skylinePoints("us")).map(p=>({...p,value:p.smooth??p.value}));
      drawLine(document.getElementById("chartSkyline"),[
        {points:skylineGlobal,color:COLORS.gold,fill:"rgba(255,215,128,.22)",width:2},
        {points:skylineUs,color:COLORS.pink,fill:"rgba(255,63,134,.16)",width:2}
      ],{dots:false,scan:true,labelAll:true,pad:{l:28,r:10,t:28,b:22}});
'''

new_draw = '''      updateSkylineHeader();
      const skylineComparison=skylineComparisonPoints();
      const skylineGlobal=skylineComparison.map(p=>({...p,value:p.global}));
      const skylineUs=skylineComparison.map(p=>({...p,value:p.us}));
      drawLine(document.getElementById("chartSkyline"),[
        {points:skylineGlobal,color:COLORS.gold,fill:"rgba(255,215,128,.22)",width:2},
        {points:skylineUs,color:COLORS.pink,fill:"rgba(255,63,134,.16)",width:2}
      ],{dots:true,scan:true,labelAll:true,pad:{l:28,r:10,t:28,b:22}});
'''

if old_draw not in text:
    raise SystemExit("STOP: expected Chart 4 drawing block not found.")
text = text.replace(old_draw, new_draw, 1)

p.write_text(text, encoding="utf-8")
print("Chart 4 raw paired PSI prepared.")
PY

git diff --check

echo
echo "Changed file only:"
git status --short
echo
git diff --stat -- "$FILE"
echo
echo "Chart 4 markers:"
grep -n -E 'skylineComparisonPoints|Global vs\. U\.S\. PSI|const skylineComparison=' "$FILE"
echo
echo "No commit or push was performed."
