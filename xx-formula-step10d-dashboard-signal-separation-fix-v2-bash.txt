#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="dashboard.html"

if [ ! -f "$FILE" ]; then
  echo "STOP: $FILE not found."
  exit 1
fi

# Refuse to operate if dashboard.html already has local edits.
if ! git diff --quiet -- "$FILE"; then
  echo "STOP: dashboard.html already has uncommitted changes."
  echo "No files were modified."
  exit 1
fi

python - <<'PY'
from pathlib import Path

p = Path("dashboard.html")
text = p.read_text(encoding="utf-8")

old_helpers = '''    function techFromScore(score){
      if(score == null) return "No PSI Match";
      if(score >= 56) return "Up";
      if(score <= 44) return "Down";
      return "Neutral";
    }
    function userDefaultFromTech(tech){
      const t = norm(tech);
      if(t.includes("up") || t.includes("bull")) return "Bullish";
      if(t.includes("down") || t.includes("bear")) return "Bearish";
      return "Neutral";
    }
'''

old_calc = '''        const score = summary.score;
        const liveTech = summary.tech || "N/A";
        const tech = liveTech !== "N/A" ? liveTech : techFromScore(score);
        const actualUser = window.PSD_USER_SENTIMENT?.[summary.name || dataKey];
        const user = actualUser && actualUser !== "N/A" ? actualUser : userDefaultFromTech(tech);
'''

new_calc = '''        const score = summary.score;
        const tech = summary.tech || "N/A";
        const actualUser = window.PSD_USER_SENTIMENT?.[summary.name || dataKey];
        const user = actualUser && actualUser !== "N/A" ? actualUser : "N/A";
'''

# PRE-FLIGHT: verify every expected block before writing anything.
errors = []

if text.count(old_helpers) != 1:
    errors.append(f"expected helper block exactly once; found {text.count(old_helpers)}")

if text.count(old_calc) != 1:
    errors.append(f"expected calcInstrument fallback block exactly once; found {text.count(old_calc)}")

if text.count("techFromScore(") != 2:
    errors.append(f"expected techFromScore definition + one call; found {text.count('techFromScore(')} occurrences")

if text.count("userDefaultFromTech(") != 2:
    errors.append(f"expected userDefaultFromTech definition + one call; found {text.count('userDefaultFromTech(')} occurrences")

if errors:
    print("STOP: pre-flight verification failed:")
    for e in errors:
        print(" -", e)
    print("No files were written.")
    raise SystemExit(1)

# Apply in memory only after all checks pass.
new_text = text.replace(old_helpers, "", 1)
new_text = new_text.replace(old_calc, new_calc, 1)

# POST-FLIGHT in memory before write.
post_errors = []

if "techFromScore(" in new_text:
    post_errors.append("techFromScore still present after planned replacement")

if "userDefaultFromTech(" in new_text:
    post_errors.append("userDefaultFromTech still present after planned replacement")

if 'const tech = summary.tech || "N/A";' not in new_text:
    post_errors.append("technical-data-only assignment missing")

if 'const user = actualUser && actualUser !== "N/A" ? actualUser : "N/A";' not in new_text:
    post_errors.append("vote-only user sentiment fallback missing")

# Preserve unrelated Deep Analysis thresholds.
for expected in [
    "return score > 58 ? 'BULL' : score < 44 ? 'BEAR' : 'BAL';",
    "const pressure = score > 58 ? 'Bullish' : score < 44 ? 'Bearish' : 'Balanced';",
    "const badge = score > 58 ? 'Risk-On' : score < 44 ? 'Risk-Off' : 'Balanced';",
]:
    if expected not in new_text:
        post_errors.append("expected Deep Analysis logic was unexpectedly altered")

if post_errors:
    print("STOP: in-memory post-flight verification failed:")
    for e in post_errors:
        print(" -", e)
    print("No files were written.")
    raise SystemExit(1)

p.write_text(new_text, encoding="utf-8")
print("Dashboard signal-separation cleanup prepared.")
PY

git diff --check

echo
echo "Changed file only:"
git status --short -- dashboard.html sentiment-core.js

echo
echo "Diff summary:"
git diff --stat -- dashboard.html

echo
echo "1) PSI-derived technical fallback removed:"
grep -nE 'techFromScore|score *>= *56.*Up|score *<= *44.*Down' dashboard.html || true

echo
echo "2) Technical direction remains sourced from shared core:"
grep -nE \
  'const tech = summary\.tech|PSDCore.*getTechnicalBias|getTechnicalForInstrument' \
  dashboard.html | head -100 || true

echo
echo "3) User Sentiment remains vote-only / N/A:"
grep -nE \
  'const actualUser|const user = actualUser|userDefaultFromTech' \
  dashboard.html | head -80 || true

echo
echo "4) Deep Analysis 58/44 logic was intentionally preserved:"
grep -nE \
  'score > 58|score < 44|Risk-On|Risk-Off|pressureLabel' \
  dashboard.html | head -100 || true

echo
echo "5) Official PSI labels still use PSDCore:"
grep -nE \
  'function label\(score\)|PSDCore.*classifySentiment|function sentimentColor|PSDCore.*sentimentColor' \
  dashboard.html | head -80 || true

echo
echo "No Deep Analysis thresholds, Heatwave, charts, PSI formulas, ribbon/header, CSS, engine data, or sentiment-core.js were modified."
echo "No commit or push was performed."
