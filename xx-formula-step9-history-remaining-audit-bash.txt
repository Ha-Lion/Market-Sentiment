#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="sentiment-history.html"
CORE="sentiment-core.js"

if [ ! -f "$FILE" ] || [ ! -f "$CORE" ]; then
  echo "STOP: required file missing."
  exit 1
fi

echo "=== HISTORICAL SENTIMENT REMAINING FORMULA AUDIT — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "1) Shared core load / PSDCore usage:"
grep -nE \
  'sentiment-core.js|PSDCore|classifySentiment|sentimentColor|badgeClass' \
  "$FILE" | head -220 || true
echo

echo "2) Remaining local 56/45 PSI classifications:"
grep -nE \
  'v[[:space:]]*>?=[[:space:]]*56|v[[:space:]]*<[[:space:]]*45|current[[:space:]]*>?=[[:space:]]*56|current[[:space:]]*<[[:space:]]*45|average[[:space:]]*>?=[[:space:]]*56|average[[:space:]]*<[[:space:]]*45|filter\(v=>v>=56|filter\(v=>v<45|v>=45&&v<56' \
  "$FILE" || true
echo

echo "3) Context around remaining classification/stat logic:"
for range in "2635,2710" "2915,2970" "3100,3160" "3210,3265"; do
  echo "----- lines $range -----"
  sed -n "${range}p" "$FILE"
  echo
done

echo "4) Bias / CSV label helper definitions and uses:"
grep -nE \
  'function bias\(|function sentimentLabelForCSV|bias\(|sentimentLabelForCSV\(' \
  "$FILE" | head -240 || true
echo

echo "5) Any official PSI thresholds elsewhere in Historical:"
grep -nE \
  '(^|[^0-9])(70|69|56|55|45|44|31|30)([^0-9]|$)' \
  "$FILE" | head -300 || true
echo

echo "6) Shared core authority:"
grep -nE \
  'const VERSION|function classifySentiment|function sentimentColor|function badgeClass|const API' \
  "$CORE" | head -120 || true
echo

echo "=== END HISTORICAL AUDIT ==="
echo "No files were modified."
