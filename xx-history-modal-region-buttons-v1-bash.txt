#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="xx-sentiment-history-lab.html"

if [ ! -f "$FILE" ]; then
  echo "STOP: $FILE not found."
  exit 1
fi

python - <<'PY'
from pathlib import Path

p = Path("xx-sentiment-history-lab.html")
text = p.read_text(encoding="utf-8")

MARKER = "xx-history-modal-region-buttons-v1"
if MARKER in text:
    raise SystemExit("STOP: modal region-button patch is already installed.")

# ------------------------------------------------------------
# 1) Add region buttons to enlarged-chart toolbar.
# ------------------------------------------------------------
old_toolbar = '''          <div class="xx-history-modal-toolbar">
            <button class="xx-history-modal-tool" type="button" data-xx-action="zoom-out">− Zoom out</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="zoom-in">+ Zoom in</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="earlier">← Earlier</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="later">Later →</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="all">All history</button>
            <span class="xx-history-modal-range" id="xxHistoryModalRange"></span>
          </div>
'''

new_toolbar = '''          <div class="xx-history-modal-toolbar">
            <div class="xx-history-modal-regions" aria-label="Region selection">
              <span class="xx-history-modal-show-label">Show</span>
              <button type="button" class="xx-region-btn" data-region="us" data-xx-modal-region="us" aria-pressed="false">U.S.A.</button>
              <button type="button" class="xx-region-btn" data-region="global" data-xx-modal-region="global" aria-pressed="false">Global</button>
              <button type="button" class="xx-region-btn" data-region="europe" data-xx-modal-region="europe" aria-pressed="false">Europe</button>
              <button type="button" class="xx-region-btn" data-region="all" data-xx-modal-region="all" aria-pressed="false">All</button>
            </div>

            <span class="xx-history-modal-separator" aria-hidden="true"></span>

            <button class="xx-history-modal-tool" type="button" data-xx-action="zoom-out">− Zoom out</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="zoom-in">+ Zoom in</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="earlier">← Earlier</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="later">Later →</button>
            <button class="xx-history-modal-tool" type="button" data-xx-action="all">All history</button>
            <span class="xx-history-modal-range" id="xxHistoryModalRange"></span>
          </div>
'''

if old_toolbar not in text:
    raise SystemExit("STOP: enlarged-chart toolbar anchor not found.")
text = text.replace(old_toolbar, new_toolbar, 1)

# ------------------------------------------------------------
# 2) Style modal region controls.
# ------------------------------------------------------------
css_anchor = '''    .xx-history-modal-tool{
      min-height:31px;
      padding:5px 10px;
'''

css_insert = '''    .xx-history-modal-regions{
      display:inline-flex;
      align-items:center;
      gap:5px;
      flex-wrap:wrap;
    }

    .xx-history-modal-show-label{
      margin-right:2px;
      color:var(--hist-muted,#94a3b8);
      font:900 10px/1 Inter,"Segoe UI",Arial,sans-serif;
      text-transform:uppercase;
      letter-spacing:.03em;
    }

    .xx-history-modal-separator{
      width:1px;
      height:26px;
      margin:0 3px;
      background:var(--hist-line,rgba(148,163,184,.24));
    }

    .xx-history-modal-regions .xx-region-btn{
      min-height:29px;
      padding:4px 9px;
      cursor:pointer;
    }

'''

if css_anchor not in text:
    raise SystemExit("STOP: modal toolbar CSS anchor not found.")
text = text.replace(css_anchor, css_insert + css_anchor, 1)

# ------------------------------------------------------------
# 3) Add modal-region click behavior.
#    Uses the same shared 'selected' Set as the main charts.
# ------------------------------------------------------------
listener_anchor = '''      modal.querySelectorAll("[data-xx-action]").forEach(btn=>{
        btn.addEventListener("click",()=>{
          xxAdjustModal(btn.dataset.xxAction);
        });
      });

      document.addEventListener("keydown",e=>{
'''

listener_new = '''      modal.querySelectorAll("[data-xx-action]").forEach(btn=>{
        btn.addEventListener("click",()=>{
          xxAdjustModal(btn.dataset.xxAction);
        });
      });

      modal.querySelectorAll("[data-xx-modal-region]").forEach(btn=>{
        btn.addEventListener("click",()=>{
          const key=btn.dataset.xxModalRegion;

          if(key==="all"){
            selected.clear();
            ["us","global","europe"].forEach(k=>selected.add(k));
          }else if(selected.has(key)){
            if(selected.size>1) selected.delete(key);
          }else{
            selected.add(key);
          }

          updateButtons();
          render();
          xxRenderHistoryModal();
        });
      });

      document.addEventListener("keydown",e=>{
'''

if listener_anchor not in text:
    raise SystemExit("STOP: modal toolbar listener anchor not found.")
text = text.replace(listener_anchor, listener_new, 1)

# ------------------------------------------------------------
# 4) Sync modal button active states whenever modal renders.
# ------------------------------------------------------------
render_anchor = '''    function xxRenderHistoryModal(hover=null){
      if(!xxHistoryModalState) return;

      const data=xxVisibleModalData();
'''

render_new = '''    function xxRenderHistoryModal(hover=null){
      if(!xxHistoryModalState) return;

      updateButtons();

      const data=xxVisibleModalData();
'''

if render_anchor not in text:
    raise SystemExit("STOP: modal render anchor not found.")
text = text.replace(render_anchor, render_new, 1)

# ------------------------------------------------------------
# 5) Marker.
# ------------------------------------------------------------
if "</head>" not in text:
    raise SystemExit("STOP: </head> not found.")
text = text.replace("</head>", f'  <!-- {MARKER} -->\n</head>', 1)

p.write_text(text, encoding="utf-8")

print("Historical Sentiment LAB enlarged-chart region buttons added.")
print("Added: U.S.A. / Global / Europe / All")
print("Buttons stay synchronized with the main chart selection.")
print("Production sentiment-history.html: UNTOUCHED")
PY

git diff --check

echo
echo "Changed files:"
git status --short -- xx-sentiment-history-lab.html sentiment-history.html

echo
echo "Verification:"
grep -n -E \
  'xx-history-modal-region-buttons-v1|data-xx-modal-region|xx-history-modal-regions|xx-history-modal-separator' \
  xx-sentiment-history-lab.html | head -40

echo
echo "No commit or push was performed."
