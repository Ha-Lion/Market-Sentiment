#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

echo "=== FORMULA HELPER CLEANUP PREVIEW — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "Formula helper .txt files that WOULD be removed:"
find . -maxdepth 1 -type f -name 'xx-formula-*.txt' -printf '%f\n' | sort
echo

COUNT="$(find . -maxdepth 1 -type f -name 'xx-formula-*.txt' | wc -l | tr -d ' ')"
echo "Total formula helper files: $COUNT"
echo

echo "Protected lab/reference files that WILL NOT be touched:"
for f in \
  xx-dashboard-lab.html \
  xx-dashboard-lab2.html \
  xx-sentiment-history-lab.html \
  xx-ribbon-lab.css \
  xx-site-nav-lab.js \
  xx-vote-widget-lab.js
do
  [ -e "$f" ] && echo "KEEP: $f"
done
echo

echo "Other xx-* files that are NOT formula helper .txt files and WILL NOT be touched:"
find . -maxdepth 1 -type f -name 'xx-*' ! -name 'xx-formula-*.txt' -printf '%f\n' | sort
echo

echo "=== END CLEANUP PREVIEW ==="
echo "No files were modified or deleted."
