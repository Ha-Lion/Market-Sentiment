#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

LAB="xx-sentiment-history-lab.html"
PROD="sentiment-history.html"

if [ ! -f "$LAB" ]; then
  echo "STOP: $LAB not found."
  exit 1
fi

if [ ! -f "$PROD" ]; then
  echo "STOP: $PROD not found."
  exit 1
fi

python - <<'PY'
from pathlib import Path
import re

lab = Path("xx-sentiment-history-lab.html")
prod = Path("sentiment-history.html")

text = lab.read_text(encoding="utf-8")

# Restore production browser title if the lab prefix is still present.
text = text.replace(
    "<title>XX Lab — Historical Market Sentiment",
    "<title>Historical Market Sentiment",
    1
)

# Remove the fixed HISTORY LAB marker only.
text, n = re.subn(
    r'\s*<div id="xx-history-lab-marker"[^>]*>HISTORY LAB</div>\s*',
    '\n',
    text,
    count=1,
    flags=re.S
)

# Fallback for the original multiline inline-style marker structure.
if n == 0:
    text, n = re.subn(
        r'\s*<div id="xx-history-lab-marker" style=".*?">HISTORY LAB</div>\s*',
        '\n',
        text,
        count=1,
        flags=re.S
    )

# Safety: production must not visibly retain the lab badge.
if 'HISTORY LAB' in text:
    raise SystemExit("STOP: HISTORY LAB marker still found after cleanup.")

prod.write_text(text, encoding="utf-8")

print("Historical Sentiment production migration prepared.")
print("Lab source: xx-sentiment-history-lab.html")
print("Production target: sentiment-history.html")
PY

git diff --check

echo
echo "Expected changed production file:"
git status --short -- sentiment-history.html xx-sentiment-history-lab.html

echo
echo "Production diff summary:"
git diff --stat -- sentiment-history.html

echo
echo "Safety checks:"
if grep -q 'HISTORY LAB' sentiment-history.html; then
  echo "ERROR: HISTORY LAB marker remains"
  exit 1
else
  echo "OK: no HISTORY LAB marker"
fi

if grep -q 'Historical Market Sentiment Dashboard' sentiment-history.html; then
  echo "OK: production dashboard title present"
else
  echo "ERROR: production dashboard title missing"
  exit 1
fi

if grep -q 'Daily Historical PSI Data' sentiment-history.html; then
  echo "OK: Daily Historical PSI Data present"
else
  echo "ERROR: Daily Historical PSI Data missing"
  exit 1
fi

if grep -q 'xx-history-final-chart-readability-v2' sentiment-history.html; then
  echo "OK: approved readability update present"
else
  echo "ERROR: approved readability update missing"
  exit 1
fi

echo
echo "Shared ribbon/theme files were not modified by this helper:"
git status --short -- site-nav.js site-theme.js site-theme.css

echo
echo "No commit or push was performed."
