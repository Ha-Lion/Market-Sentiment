#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

echo "=== FINAL FORMULA CERTIFICATION — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "1) Shared PSI authority:"
grep -nE \
  'const VERSION|function classifySentiment|function sentimentColor|function calculatePSI|function buildInstrumentSummary' \
  sentiment-core.js || true
echo

echo "2) Remaining hard-coded OFFICIAL PSI thresholds outside shared core:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'score[[:space:]]*(>=|>|<=|<)[[:space:]]*(70|56|45|31|30|44|55|69)|s[[:space:]]*(>=|>|<=|<)[[:space:]]*(70|56|45|31|30|44|55|69)|v[[:space:]]*(>=|>|<=|<)[[:space:]]*(70|56|45|31|30|44|55|69)' \
  . || true
echo

echo "3) Old PSI -> Technical / Technical -> User fallback logic:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'techFromScore|userDefaultFromTech|falls back to daily technical direction' \
  . || true
echo

echo "4) Static official PSI range text still embedded in production pages:"
grep -RniE \
  --include='*.html' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  '0[[:space:]]*(to|–|-)[[:space:]]*30|31[[:space:]]*(to|–|-)[[:space:]]*44|45[[:space:]]*(to|–|-)[[:space:]]*55|56[[:space:]]*(to|–|-)[[:space:]]*69|70[[:space:]]*(to|–|-)[[:space:]]*100' \
  . || true
echo

echo "5) PSI guide central linkage:"
grep -nE \
  'sentiment-core\.js|psiRangeList|PSDCore\.classifySentiment|Technical direction and user sentiment are displayed separately' \
  public-sentiment-index.html || true
echo

echo "6) Dashboard central linkage + intentional separate Deep Analysis model:"
grep -nE \
  'PSDCore\.classifySentiment|psiBandsFromCore|renderHeatLegend|score > 58|score < 44|Risk-On|Risk-Off' \
  dashboard.html | head -120 || true
echo

echo "7) PDF report shared classification:"
grep -nE \
  'PSDCore\.classifySentiment\(score\)|const color = category' \
  pdf-report.js || true
echo

echo "8) Production pages loading sentiment-core.js:"
for f in ./*.html; do
  base="$(basename "$f")"
  [[ "$base" == xx-* ]] && continue
  [[ "$base" == *-lab.html ]] && continue
  if grep -q 'sentiment-core.js' "$f"; then
    echo "$base"
  fi
done | sort
echo

echo "=== END FINAL CERTIFICATION ==="
echo "No files were modified."
