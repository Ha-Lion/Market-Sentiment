assets = config.get("assets") or []
by_id = {str(a.get("id")): a for a in assets if a.get("id") is not None}

def unique(vals):
    out=[]
    seen=set()
    for v in vals or []:
        s=str(v).strip()
        if not s:
            continue
        k=s.lower()
        if k in seen:
            continue
        seen.add(k)
        out.append(s)
    return out

def from_config(asset):
    name = asset.get("display_name") or asset.get("name") or asset.get("symbol")
    data_key = asset.get("data_key") or asset.get("name") or name
    symbol = asset.get("symbol") or name
    aliases = unique([
        *(asset.get("aliases") or []),
        name, data_key, asset.get("name"), symbol
    ])
    history = unique([
        data_key, name, asset.get("name"), symbol,
        *(asset.get("history_keys") or [])
    ])
    return {
        "name":name,
        "dataKey":data_key,
        "symbol":symbol,
        "kind":asset.get("type") or "asset",
        "aliases":aliases,
        "historyKeys":history,
        "official":True,
    }

def js_str(s):
    return json.dumps(str(s), ensure_ascii=False)

def js_item(item):
    return (
        "{name:"+js_str(item["name"])+
        ", dataKey:"+js_str(item.get("dataKey") or item["name"])+
        ", symbol:"+js_str(item.get("symbol") or item["name"])+
        ", kind:"+js_str(item.get("kind") or "asset")+
        ", aliases:"+json.dumps(item.get("aliases") or [], ensure_ascii=False)+
        ", historyKeys:"+json.dumps(item.get("historyKeys") or [], ensure_ascii=False)+
        ", official:"+("true" if item.get("official") is True else "false")+"}"
    )

def make_lab_items(cat, plan):
    existing=[]
    for asset_id in plan["existing_ids"]:
        asset=by_id.get(asset_id)
        if not asset:
            raise SystemExit(f"STOP: assets_config missing expected id {asset_id} for {cat}")
        existing.append(from_config(asset))

    if cat=="crypto":
        idx=plan["existing_ids"].index(plan["insert_before"])
        items=existing[:idx] + plan["new"] + existing[idx:]
    else:
        items=existing + plan["new"]

    if len(items) != plan["target"]:
        raise SystemExit(f"STOP: {cat} target mismatch: built {len(items)}, expected {plan['target']}")
    return items

def set_noindex(text):
    if re.search(r'<meta\s+name=["\']robots["\']', text, re.I):
        text=re.sub(
            r'<meta\s+name=["\']robots["\'][^>]*>',
            '<meta name="robots" content="noindex, nofollow">',
            text, count=1, flags=re.I
        )
    else:
        text=text.replace("<head>",'<head>\n  <meta name="robots" content="noindex, nofollow">',1)
    text=re.sub(r'<link\s+rel=["\']canonical["\'][^>]*>','',text,count=1,flags=re.I)
    return text

def set_title(text,title):
    return re.sub(r'<title>.*?</title>',f'<title>{title}</title>',text,count=1,flags=re.I|re.S)

def add_lab_marker(text,label):
    if "xx-category-lab-marker" in text:
        return text
    marker=(
        '<div id="xx-category-lab-marker" style="'
        'position:fixed;right:12px;bottom:12px;z-index:99999;padding:5px 9px;'
        'border-radius:999px;font:700 10px/1.1 Inter,Arial,sans-serif;letter-spacing:.08em;'
        'background:rgba(15,23,42,.94);color:#f6c85f;border:1px solid rgba(246,200,95,.48);'
        'pointer-events:none;">'+label+'</div>'
    )
    b=text.find("<body")
    if b<0:
        raise SystemExit("STOP: <body> not found in lab source")
    pos=text.find(">",b)+1
    return text[:pos]+"\n"+marker+text[pos:]

def make_items_mutable(text,varname):
    pat=r'\bconst\s+'+re.escape(varname)+r'\s*='
    if re.search(pat,text):
        text=re.sub(pat,'let '+varname+' =',text,count=1)
    return text

def remove_buildrows_cap(text):
    m=re.search(r'function\s+buildRows\s*\(\)\s*\{',text)
    if not m:
        raise SystemExit("STOP: function buildRows() not found")
    next_fun=re.search(r'\n\s*function\s+\w+\s*\(',text[m.end():])
    if not next_fun:
        raise SystemExit("STOP: could not find function after buildRows()")
    end=m.end()+next_fun.start()
    block=text[m.start():end]
    new_block=re.sub(r'\.slice\(\s*0\s*,\s*\d+\s*\)\s*;',';',block,count=1)
    return text[:m.start()]+new_block+text[end:] if new_block!=block else text

def inject_lab_universe(text,varname,items):
    marker=f"XX_{varname}_LAB_ITEMS"
    js="    const "+marker+" = [\n      "+",\n      ".join(js_item(x) for x in items)+"\n    ];\n\n"
    anchor=re.search(r'\n\s*async\s+function\s+init\s*\(\)\s*\{',text)
    if not anchor:
        raise SystemExit("STOP: async function init() anchor not found")
    text=text[:anchor.start()+1]+js+text[anchor.start()+1:]

    needle="        applyAssetConfig(assetConfig);"
    if needle not in text:
        raise SystemExit("STOP: applyAssetConfig(assetConfig) init anchor not found")
    text=text.replace(
        needle,
        needle+f"\n        {varname} = {marker}.map(item => ({{...item}}));",
        1
    )
    return text

def inject_daily_technical(text,varname):
    if "function dailyTechnicalDirection(" in text:
        return text

    fn=f'''    function dailyTechnicalDirection(name){{
      const item={varname}.find(x=>x.name===name);
      if(!item)return "N/A";

      const root=(technicalData && (technicalData.technical || technicalData)) || {{}};
      const keys=[
        item.dataKey,item.name,item.symbol,
        ...(Array.isArray(item.historyKeys)?item.historyKeys:[])
      ].filter(Boolean).map(x=>normalizeText(x));

      const foundKey=Object.keys(root).find(key=>keys.includes(normalizeText(key)));
      const entry=foundKey ? root[foundKey] : null;
      const daily=entry && (entry.daily || entry.Daily);

      let raw="";
      if(typeof daily==="string") raw=daily;
      else if(daily) raw=daily.direction || daily.signal || daily.label || daily.trend || daily.value || "";

      raw=String(raw||"").trim();
      if(!raw || raw.toUpperCase()==="N/A") return "N/A";

      const n=normalizeText(raw);
      if(n.includes("bull") || n.includes("buy") || n.includes("up")) return "Bullish";
      if(n.includes("bear") || n.includes("sell") || n.includes("down")) return "Bearish";
      if(n.includes("neutral") || n.includes("mixed") || n.includes("flat")) return "Neutral";
      return raw;
    }}

'''
    anchor=re.search(r'\n\s*function\s+renderTopCards\s*\(\)\s*\{',text)
    if not anchor:
        raise SystemExit("STOP: renderTopCards() anchor not found")
    text=text[:anchor.start()+1]+fn+text[anchor.start()+1:]

    lines=text.splitlines()
    out=[]
    inserted=0
    for line in lines:
        out.append(line)
        if (
            'class="market-insight"' in line
            and 'marketDirection(row.name)' in line
            and 'Daily technical direction' not in line
        ):
            indent=line[:len(line)-len(line.lstrip())]
            out.append(indent+'<div class="market-insight"><span>Daily technical direction</span><b>${esc(dailyTechnicalDirection(row.name))}</b></div>')
            inserted+=1
    if inserted != 1:
        raise SystemExit(f"STOP: expected 1 Daily Market Insight card anchor, found {inserted}")
    return "\n".join(out)+("\n" if text.endswith("\n") else "")

def verify_lab(path,varname,target):
    text=path.read_text(encoding="utf-8")
    marker=f"XX_{varname}_LAB_ITEMS"
    m=re.search(r'const\s+'+re.escape(marker)+r'\s*=\s*\[(.*?)\n\s*\];',text,re.S)
    if not m:
        raise SystemExit(f"STOP: could not verify {marker}")
    count=len(re.findall(r'\{name:',m.group(1)))
    if count!=target:
        raise SystemExit(f"STOP: {path.name} has {count} lab items, expected {target}")
    if "Daily technical direction" not in text:
        raise SystemExit(f"STOP: {path.name} missing Daily technical direction")
    if 'noindex, nofollow' not in text:
        raise SystemExit(f"STOP: {path.name} missing noindex")
    return count

    text=remove_buildrows_cap(text)

    lab.write_text(text,encoding="utf-8")
    verify_lab(lab,plan["var"],plan["target"])
    created.append((cat,lab,items))
