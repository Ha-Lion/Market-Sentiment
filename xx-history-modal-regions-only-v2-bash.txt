#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="xx-sentiment-history-lab.html"

if [ ! -f "$FILE" ]; then
  echo "STOP: $FILE not found."
  exit 1
fi

python - <<'PY'
from pathlib import Path
import re

p = Path("xx-sentiment-history-lab.html")
text = p.read_text(encoding="utf-8")

MARKER = "xx-history-modal-regions-only-v2"
if MARKER in text:
    raise SystemExit("STOP: modal regions-only patch is already installed.")

# ------------------------------------------------------------
# 1) Replace the enlarged-chart toolbar contents.
#    Keep ONLY region controls + range text.
#    Wheel/trackpad zoom/pan remains enabled.
# ------------------------------------------------------------
pattern = re.compile(
    r'''          <div class="xx-history-modal-toolbar">\n'''
    r'''.*?'''
    r'''            <span class="xx-history-modal-range" id="xxHistoryModalRange"></span>\n'''
    r'''          </div>''',
    re.S
)

replacement = '''          <div class="xx-history-modal-toolbar">
            <div class="xx-history-modal-regions" aria-label="Region selection">
              <span class="xx-history-modal-show-label">Show</span>
              <button type="button" class="xx-region-btn" data-region="us" data-xx-modal-region="us" aria-pressed="false">U.S.A.</button>
              <button type="button" class="xx-region-btn" data-region="global" data-xx-modal-region="global" aria-pressed="false">Global</button>
              <button type="button" class="xx-region-btn" data-region="europe" data-xx-modal-region="europe" aria-pressed="false">Europe</button>
              <button type="button" class="xx-region-btn" data-region="all" data-xx-modal-region="all" aria-pressed="false">All</button>
            </div>
            <span class="xx-history-modal-range" id="xxHistoryModalRange"></span>
          </div>'''

text, n = pattern.subn(replacement, text, count=1)
if n != 1:
    raise SystemExit("STOP: enlarged-chart toolbar block not found.")

# ------------------------------------------------------------
# 2) Ensure modal-region CSS exists.
# ------------------------------------------------------------
if ".xx-history-modal-regions{" not in text:
    css_anchor = '''    .xx-history-modal-tool{
'''
    css_insert = '''    .xx-history-modal-regions{
      display:inline-flex;
      align-items:center;
      gap:6px;
      flex-wrap:wrap;
    }

    .xx-history-modal-show-label{
      margin-right:2px;
      color:var(--hist-muted,#94a3b8);
      font:900 10px/1 Inter,"Segoe UI",Arial,sans-serif;
      text-transform:uppercase;
      letter-spacing:.03em;
    }

    .xx-history-modal-regions .xx-region-btn{
      min-height:29px;
      padding:4px 10px;
      cursor:pointer;
    }

'''
    if css_anchor not in text:
        raise SystemExit("STOP: modal CSS anchor not found.")
    text = text.replace(css_anchor, css_insert + css_anchor, 1)

# ------------------------------------------------------------
# 3) Ensure the modal region buttons have click behavior.
# ------------------------------------------------------------
if 'modal.querySelectorAll("[data-xx-modal-region]")' not in text:
    listener_anchor = '''      modal.querySelectorAll("[data-xx-action]").forEach(btn=>{
        btn.addEventListener("click",()=>{
          xxAdjustModal(btn.dataset.xxAction);
        });
      });

'''
    listener_new = listener_anchor + '''      modal.querySelectorAll("[data-xx-modal-region]").forEach(btn=>{
        btn.addEventListener("click",()=>{
          const key=btn.dataset.xxModalRegion;

          if(key==="all"){
            selected.clear();
            ["us","global","europe"].forEach(k=>selected.add(k));
          }else if(selected.has(key)){
            if(selected.size>1) selected.delete(key);
          }else{
            selected.add(key);
          }

          updateButtons();
          render();
          xxRenderHistoryModal();
        });
      });

'''
    if listener_anchor not in text:
        raise SystemExit("STOP: modal action-listener anchor not found.")
    text = text.replace(listener_anchor, listener_new, 1)

# ------------------------------------------------------------
# 4) Make sure active states sync before every modal render.
# ------------------------------------------------------------
render_anchor = '''    function xxRenderHistoryModal(hover=null){
      if(!xxHistoryModalState) return;

'''
if "function xxRenderHistoryModal(hover=null)" not in text:
    raise SystemExit("STOP: modal render function not found.")

render_window = text[text.index("    function xxRenderHistoryModal(hover=null){"):
                     text.index("    function xxAdjustModal", text.index("    function xxRenderHistoryModal(hover=null){"))]
if "updateButtons();" not in render_window:
    text = text.replace(render_anchor, render_anchor + "      updateButtons();\n\n", 1)

# ------------------------------------------------------------
# 5) Marker.
# ------------------------------------------------------------
if "</head>" not in text:
    raise SystemExit("STOP: </head> not found.")
text = text.replace("</head>", f'  <!-- {MARKER} -->\n</head>', 1)

p.write_text(text, encoding="utf-8")

print("Historical Sentiment LAB enlarged toolbar simplified.")
print("Visible toolbar controls: U.S.A. / Global / Europe / All")
print("Removed visible: Zoom out / Zoom in / Earlier / Later / All history")
print("Mouse wheel + trackpad zoom/pan remains enabled.")
print("Production sentiment-history.html: UNTOUCHED")
PY

git diff --check

echo
echo "Changed files:"
git status --short -- xx-sentiment-history-lab.html sentiment-history.html

echo
echo "Verification:"
grep -n -E \
  'xx-history-modal-regions-only-v2|data-xx-modal-region|Zoom out|Zoom in|All history|xx-history-modal-range' \
  xx-sentiment-history-lab.html | head -50

echo
echo "No commit or push was performed."
