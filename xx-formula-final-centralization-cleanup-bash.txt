#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILES=(
  index.html
  pdf-report.js
  dashboard.html
  public-sentiment-index.html
  how-it-works.html
)

for f in "${FILES[@]}"; do
  if [ ! -f "$f" ]; then
    echo "STOP: required file missing: $f"
    exit 1
  fi
  if ! git diff --quiet -- "$f"; then
    echo "STOP: $f already has uncommitted changes."
    echo "No files were modified."
    exit 1
  fi
done

python - <<'PY'
from pathlib import Path

files = {
    name: Path(name).read_text(encoding="utf-8")
    for name in [
        "index.html",
        "pdf-report.js",
        "dashboard.html",
        "public-sentiment-index.html",
        "how-it-works.html",
    ]
}

errors = []

# ---------- HOME ----------
home_old = '''    function scoreLabel(score){
      if(window.PSDCore && window.PSDCore.classifySentiment){
        return window.PSDCore.classifySentiment(score);
      }

      if(score >= 70) return "Strong Bullish";
      if(score >= 56) return "Bullish";
      if(score >= 45) return "Mixed/Neutral";
      if(score >= 31) return "Bearish";

      return "Strong Bearish";
    }

    function scoreColor(score){
      const s = Number(score);
      if(!Number.isFinite(s)) return "var(--gold)";
      if(s >= 70) return "var(--green)";
      if(s >= 56) return "var(--green)";
      if(s >= 45) return "var(--gold)";
      if(s >= 31) return "var(--orange)";

      return "var(--red)";
    }
'''

home_new = '''    function scoreLabel(score){
      if(window.PSDCore && window.PSDCore.classifySentiment){
        return window.PSDCore.classifySentiment(score);
      }
      return "No PSI Match";
    }

    function scoreColor(score){
      const s = Number(score);
      if(!Number.isFinite(s)) return "var(--gold)";
      const category = scoreLabel(s);
      if(category === "Strong Bullish" || category === "Bullish") return "var(--green)";
      if(category === "Mixed/Neutral") return "var(--gold)";
      if(category === "Bearish") return "var(--orange)";
      if(category === "Strong Bearish") return "var(--red)";
      return "var(--gold)";
    }
'''

if files["index.html"].count(home_old) != 1:
    errors.append("Home scoreLabel/scoreColor exact block was not found exactly once.")

# ---------- PDF ----------
pdf_old = '    const color = score >= 70 ? "#3fb950" : score <= 30 ? "#f85149" : "#d29922";'
pdf_new = '''    const category = window.PSDCore && typeof window.PSDCore.classifySentiment === "function"
      ? window.PSDCore.classifySentiment(score)
      : label;
    const color = category === "Strong Bullish" ? "#3fb950" : category === "Strong Bearish" ? "#f85149" : "#d29922";'''

if files["pdf-report.js"].count(pdf_old) != 1:
    errors.append("PDF hard-coded PSI color line was not found exactly once.")

# ---------- DASHBOARD HEATWAVE ----------
dash_old = '''        <div class="heat-legend" aria-label="Heatwave color scale">
          <span><i style="background:#ef4444;color:#ef4444"></i>0–30 Bearish</span>
          <span><i style="background:#facc15;color:#facc15"></i>31–44 Mixed Bearish</span>
          <span><i style="background:#3b82f6;color:#3b82f6"></i>45–55 Neutral</span>
          <span><i style="background:#2dd4bf;color:#2dd4bf"></i>56–69 Mixed Bullish</span>
          <span><i style="background:#16a34a;color:#16a34a"></i>70–100 Bullish</span>
        </div>
        <div class="heatmap" id="heatmapGrid"></div>
        <div class="preview-note">Heatwave reads each market's recent PSI-style values. Daily shows recent hours, Weekly shows recent days, and Monthly shows sampled historical points. Color scale: 0–30 Red/Bearish, 31–44 Yellow/Mixed Bearish, 45–55 Blue/Neutral, 56–69 Light Blue-Green/Mixed Bullish, 70–100 Strong Green/Bullish.</div>'''

dash_new = '''        <div class="heat-legend" id="heatLegend" aria-label="Heatwave color scale"></div>
        <div class="heatmap" id="heatmapGrid"></div>
        <div class="preview-note" id="heatPreviewNote">Heatwave ranges follow the current shared PSI classification.</div>'''

if files["dashboard.html"].count(dash_old) != 1:
    errors.append("Dashboard static Heatwave legend block was not found exactly once.")

dash_anchor = '    function lineColor(idx){return LINE_COLORS[idx % LINE_COLORS.length]}'

dash_helpers = '''    function psiBandsFromCore(){
      if(!window.PSDCore?.classifySentiment) return [];
      const bands=[];
      let start=0;
      let current=label(0);
      for(let score=1;score<=100;score++){
        const next=label(score);
        if(next!==current){
          bands.push({min:start,max:score-1,label:current});
          start=score;
          current=next;
        }
      }
      bands.push({min:start,max:100,label:current});
      return bands;
    }
    function heatLegendLabel(category){
      if(category==="Strong Bearish") return "Bearish";
      if(category==="Bearish") return "Mixed Bearish";
      if(category==="Mixed/Neutral") return "Neutral";
      if(category==="Bullish") return "Mixed Bullish";
      if(category==="Strong Bullish") return "Bullish";
      return category;
    }
    function renderHeatLegend(){
      const bands=psiBandsFromCore();
      const legend=document.getElementById("heatLegend");
      const note=document.getElementById("heatPreviewNote");
      if(!bands.length){
        if(legend) legend.textContent="PSI color scale unavailable";
        return;
      }
      if(legend){
        legend.innerHTML=bands.map(b=>{
          const mid=Math.round((b.min+b.max)/2);
          const color=heatColor(mid);
          return `<span><i style="background:${color};color:${color}"></i>${b.min}–${b.max} ${heatLegendLabel(b.label)}</span>`;
        }).join("");
      }
      if(note){
        const scale=bands.map(b=>`${b.min}–${b.max} ${heatLegendLabel(b.label)}`).join(", ");
        note.textContent=`Heatwave reads each market's recent PSI-style values. Daily shows recent hours, Weekly shows recent days, and Monthly shows sampled historical points. Color scale: ${scale}.`;
      }
    }
    renderHeatLegend();

    function lineColor(idx){return LINE_COLORS[idx % LINE_COLORS.length]}'''

if files["dashboard.html"].count(dash_anchor) != 1:
    errors.append("Dashboard lineColor anchor was not found exactly once.")

# ---------- PSI GUIDE ----------
psi_ranges_old = '''        <ul>
          <li><b>0 to 30:</b> Bearish public sentiment.</li>
          <li><b>31 to 44:</b> Mixed / bearish sentiment.</li>
          <li><b>45 to 55:</b> Mixed or neutral sentiment.</li>
          <li><b>56 to 69:</b> Mixed / bullish sentiment.</li>
          <li><b>70 to 100:</b> Bullish public sentiment.</li>
        </ul>'''

psi_ranges_new = '''        <ul id="psiRangeList">
          <li>Loading current PSI ranges…</li>
        </ul>'''

if files["public-sentiment-index.html"].count(psi_ranges_old) != 1:
    errors.append("PSI guide static range list was not found exactly once.")

faq_old = '"text":"The Public Sentiment Index is an AI-assisted market sentiment reading that organizes public financial headlines, market narratives, technical direction, and user voting into a simple sentiment score."'
faq_new = '"text":"The Public Sentiment Index is an AI-assisted market sentiment reading derived from public financial headlines and market narratives. Technical direction and user sentiment are displayed separately and are not part of the PSI calculation."'

if files["public-sentiment-index.html"].count(faq_old) != 1:
    errors.append("PSI guide FAQ definition was not found exactly once.")

psi_body_script = '''  <script src="sentiment-core.js?v=5"></script>
  <script>
    (function(){
      const list=document.getElementById("psiRangeList");
      if(!list) return;
      if(!window.PSDCore || typeof window.PSDCore.classifySentiment!=="function"){
        list.innerHTML="<li>Current PSI ranges are temporarily unavailable.</li>";
        return;
      }
      const bands=[];
      let start=0;
      let current=window.PSDCore.classifySentiment(0);
      for(let score=1;score<=100;score++){
        const next=window.PSDCore.classifySentiment(score);
        if(next!==current){
          bands.push({min:start,max:score-1,label:current});
          start=score;
          current=next;
        }
      }
      bands.push({min:start,max:100,label:current});
      list.innerHTML=bands.map(b=>`<li><b>${b.min} to ${b.max}:</b> ${b.label} public sentiment.</li>`).join("");
    })();
  </script>
'''

if files["public-sentiment-index.html"].count("</body>") != 1:
    errors.append("PSI guide closing body tag was not found exactly once.")
if 'sentiment-core.js' in files["public-sentiment-index.html"]:
    errors.append("PSI guide already loads sentiment-core.js; refusing duplicate insertion.")

# ---------- HOW IT WORKS ----------
how_old = '        <p>Users can vote bullish or bearish on instruments without registration. If voting data is not strong enough yet, the site falls back to daily technical direction so the display stays useful.</p>'
how_new = '        <p>Users can vote bullish or bearish on instruments without registration. User Sentiment is kept separate from PSI and Technical Direction; when no user vote is available, the display remains unavailable rather than substituting another signal.</p>'

if files["how-it-works.html"].count(how_old) != 1:
    errors.append("How It Works outdated User Sentiment fallback text was not found exactly once.")

# ---------- ALL PREFLIGHTS MUST PASS ----------
if errors:
    print("STOP: pre-flight verification failed:")
    for e in errors:
        print(" -", e)
    print("No files were written.")
    raise SystemExit(1)

# Build all changes in memory.
new = dict(files)

new["index.html"] = new["index.html"].replace(home_old, home_new, 1)

new["pdf-report.js"] = new["pdf-report.js"].replace(pdf_old, pdf_new, 1)
new["pdf-report.js"] = new["pdf-report.js"].replace(
    "Uses live page values; dashboard/history/news pages should source shared PSI_CORE_V1 logic.",
    "Uses live page values; PSI classification should follow the shared PSDCore authority.",
    1
)

new["dashboard.html"] = new["dashboard.html"].replace(dash_old, dash_new, 1)
new["dashboard.html"] = new["dashboard.html"].replace(dash_anchor, dash_helpers, 1)

new["public-sentiment-index.html"] = new["public-sentiment-index.html"].replace(psi_ranges_old, psi_ranges_new, 1)
new["public-sentiment-index.html"] = new["public-sentiment-index.html"].replace(faq_old, faq_new, 1)
new["public-sentiment-index.html"] = new["public-sentiment-index.html"].replace(
    "</body>",
    psi_body_script + "</body>",
    1
)

new["how-it-works.html"] = new["how-it-works.html"].replace(how_old, how_new, 1)

# Post-flight: validate the intended result before writing anything.
post_errors = []

for needle in [
    'if(score >= 70) return "Strong Bullish";',
    'if(score >= 56) return "Bullish";',
    'if(score >= 45) return "Mixed/Neutral";',
    'if(score >= 31) return "Bearish";',
    'if(s >= 70) return "var(--green)";',
]:
    if needle in new["index.html"]:
        post_errors.append(f"Home threshold remains: {needle}")

if pdf_old in new["pdf-report.js"]:
    post_errors.append("PDF hard-coded score threshold remains.")
if "PSDCore.classifySentiment(score)" not in new["pdf-report.js"]:
    post_errors.append("PDF shared PSI classification call missing.")

for needle in ["0–30 Bearish", "31–44 Mixed Bearish", "56–69 Mixed Bullish", "70–100 Bullish"]:
    if needle in new["dashboard.html"]:
        post_errors.append(f"Dashboard static Heatwave range remains: {needle}")
if "function psiBandsFromCore()" not in new["dashboard.html"]:
    post_errors.append("Dashboard dynamic PSI band builder missing.")

for expected in [
    "return score > 58 ? 'BULL' : score < 44 ? 'BEAR' : 'BAL';",
    "const pressure = score > 58 ? 'Bullish' : score < 44 ? 'Bearish' : 'Balanced';",
    "const badge = score > 58 ? 'Risk-On' : score < 44 ? 'Risk-Off' : 'Balanced';",
]:
    if expected not in new["dashboard.html"]:
        post_errors.append("Intentional Deep Analysis 58/44 logic was unexpectedly altered.")

if '<script src="sentiment-core.js?v=5"></script>' not in new["public-sentiment-index.html"]:
    post_errors.append("PSI guide shared core script missing.")
if "0 to 30:" in new["public-sentiment-index.html"] or "70 to 100:" in new["public-sentiment-index.html"]:
    post_errors.append("PSI guide static threshold text remains.")
if "Technical direction and user sentiment are displayed separately" not in new["public-sentiment-index.html"]:
    post_errors.append("PSI guide corrected headline-only definition missing.")

if "falls back to daily technical direction" in new["how-it-works.html"]:
    post_errors.append("How It Works still describes Technical fallback for User Sentiment.")

if post_errors:
    print("STOP: in-memory post-flight verification failed:")
    for e in post_errors:
        print(" -", e)
    print("No files were written.")
    raise SystemExit(1)

for name, text in new.items():
    Path(name).write_text(text, encoding="utf-8")

print("Final website formula centralization cleanup prepared.")
PY

git diff --check
