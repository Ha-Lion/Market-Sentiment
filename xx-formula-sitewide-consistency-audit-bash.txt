#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

echo "=== SITE-WIDE PSI FORMULA CONSISTENCY SWEEP — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "1) Production HTML pages that load sentiment-core.js:"
for f in ./*.html; do
  base="$(basename "$f")"
  [[ "$base" == xx-* ]] && continue
  if grep -q 'sentiment-core.js' "$f"; then
    echo "$base"
  fi
done | sort
echo

echo "2) Production HTML pages with PSI/sentiment logic but NO sentiment-core.js:"
for f in ./*.html; do
  base="$(basename "$f")"
  [[ "$base" == xx-* ]] && continue
  if grep -qiE '\bpsi\b|sentiment_label|classifySentiment|headline_psi|bullish|bearish|mixed.?neutral' "$f" \
     && ! grep -q 'sentiment-core.js' "$f"; then
    echo "----- $base -----"
    grep -nEi '\bpsi\b|sentiment_label|classifySentiment|headline_psi|bullish|bearish|mixed.?neutral' "$f" | head -80 || true
    echo
  fi
done
echo

echo "3) Hard-coded official PSI thresholds outside sentiment-core.js:"
grep -RniE \
  --include='*.html' --include='*.js' \
  --exclude='sentiment-core.js' --exclude='glossary.js' \
  --exclude='xx-*' \
  '(>=|<=|>|<)[[:space:]]*(70|69|56|55|45|44|31|30)|\b(70|56|45|31)\b.*(Bullish|Bearish|Neutral)|(?:Bullish|Bearish|Neutral).*\b(70|56|45|31)\b' \
  . | head -320 || true
echo

echo "4) Direct sentiment_label display/use outside shared core:"
grep -RniE \
  --include='*.html' --include='*.js' \
  --exclude='sentiment-core.js' --exclude='xx-*' \
  'sentiment_label' \
  . | head -320 || true
echo

echo "5) Local classify/label helpers outside shared core:"
grep -RniE \
  --include='*.html' --include='*.js' \
  --exclude='sentiment-core.js' --exclude='xx-*' \
  'function .*classif|function .*sentiment.*label|const .*sentiment.*label.*=>|function .*bias.*\(' \
  . | head -320 || true
echo

echo "6) Potential PSI recalculation outside shared core:"
grep -RniE \
  --include='*.html' --include='*.js' \
  --exclude='sentiment-core.js' --exclude='xx-*' \
  'headline_psi_score|average.*psi|psi.*average|50[[:space:]]*\+[^\n]*score|weighted.*psi|psi.*weighted' \
  . | head -360 || true
echo

echo "7) Shared core authority:"
grep -nE \
  'const VERSION|function classifySentiment|function calculatePSI|function getTechnicalBias|function buildInstrumentSeries|function buildInstrumentPeriodScore|const API' \
  sentiment-core.js | head -180 || true
echo

echo "=== END SITE-WIDE SWEEP ==="
echo "No files were modified."
