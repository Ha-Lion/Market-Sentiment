#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

python - <<'PY'
from pathlib import Path
import re

FILES = [
    "xx-forex-sentiment-lab.html",
    "xx-crypto-lab.html",
    "xx-indices-lab.html",
    "xx-energy-lab.html",
    "xx-precious-metals-lab.html",
    "xx-ai-assets-lab.html",
    "xx-policy-assets-lab.html",
]

STYLE_BLOCK = '''
  <style id="xx-daily-tech-colors-v1">
    .daily-tech-value{
      display:inline-flex;
      align-items:center;
      justify-content:center;
      min-width:74px;
      padding:2px 7px;
      border-radius:999px;
      font-weight:800;
      line-height:1.2;
      border:1px solid transparent;
    }
    .daily-tech-value.tech-bullish{
      color:#16a34a;
      background:rgba(34,197,94,.11);
      border-color:rgba(34,197,94,.28);
    }
    .daily-tech-value.tech-bearish{
      color:#dc2626;
      background:rgba(239,68,68,.11);
      border-color:rgba(239,68,68,.28);
    }
    .daily-tech-value.tech-neutral{
      color:#b7791f;
      background:rgba(245,158,11,.11);
      border-color:rgba(245,158,11,.28);
    }
    .daily-tech-value.tech-na{
      color:#64748b;
      background:rgba(100,116,139,.10);
      border-color:rgba(100,116,139,.22);
    }
    [data-theme="dark"] .daily-tech-value.tech-bullish,
    .dark .daily-tech-value.tech-bullish{ color:#4ade80; }
    [data-theme="dark"] .daily-tech-value.tech-bearish,
    .dark .daily-tech-value.tech-bearish{ color:#f87171; }
    [data-theme="dark"] .daily-tech-value.tech-neutral,
    .dark .daily-tech-value.tech-neutral{ color:#fbbf24; }
    [data-theme="dark"] .daily-tech-value.tech-na,
    .dark .daily-tech-value.tech-na{ color:#94a3b8; }
  </style>
'''

TONE_FN = '''
    function dailyTechnicalTone(name){
      const value=String(dailyTechnicalDirection(name)||"N/A").trim().toLowerCase();
      if(value.includes("bull") || value.includes("buy") || value.includes("up")) return "tech-bullish";
      if(value.includes("bear") || value.includes("sell") || value.includes("down")) return "tech-bearish";
      if(value.includes("neutral") || value.includes("mixed") || value.includes("flat")) return "tech-neutral";
      return "tech-na";
    }

'''

for filename in FILES:
    path = Path(filename)
    if not path.exists():
        raise SystemExit(f"STOP: missing lab file: {filename}")

# Preflight all files before changing any.
for filename in FILES:
    text = Path(filename).read_text(encoding="utf-8")
    if "xx-daily-tech-colors-v1" in text:
        raise SystemExit(f"STOP: {filename} already has V1 technical colors. Nothing changed.")
    if "function dailyTechnicalDirection(" not in text:
        raise SystemExit(f"STOP: {filename} missing dailyTechnicalDirection(). Nothing changed.")
    if "Daily technical direction" not in text:
        raise SystemExit(f"STOP: {filename} missing Daily technical direction card row. Nothing changed.")
    if "</head>" not in text:
        raise SystemExit(f"STOP: {filename} missing </head>. Nothing changed.")

changed=[]

for filename in FILES:
    path = Path(filename)
    text = path.read_text(encoding="utf-8")

    text = text.replace("</head>", STYLE_BLOCK + "\n</head>", 1)

    anchor = re.search(r'\n\s*function\s+renderTopCards\s*\(\)\s*\{', text)
    if not anchor:
        raise SystemExit(f"STOP: renderTopCards() not found in {filename}")
    text = text[:anchor.start()+1] + TONE_FN + text[anchor.start()+1:]

    old = '<div class="market-insight"><span>Daily technical direction</span><b>${esc(dailyTechnicalDirection(row.name))}</b></div>'
    new = '<div class="market-insight"><span>Daily technical direction</span><b class="daily-tech-value ${dailyTechnicalTone(row.name)}">${esc(dailyTechnicalDirection(row.name))}</b></div>'

    count = text.count(old)
    if count != 1:
        raise SystemExit(f"STOP: expected exactly 1 Daily Technical Direction row in {filename}, found {count}")

    text = text.replace(old, new, 1)
    path.write_text(text, encoding="utf-8")
    changed.append(filename)

print("Daily Technical Direction color coding applied successfully.")
print()
print("Color key:")
print("  Bullish  = green")
print("  Bearish  = red")
print("  Neutral  = gold")
print("  N/A      = gray")
print()
for f in changed:
    print("  UPDATED", f)
print()
print("Production pages: UNTOUCHED")
print("assets_config.json: UNTOUCHED")
print("Engine V2: UNTOUCHED")
PY

git diff --check

echo
echo "Changed lab files:"
git status --short -- \
  xx-forex-sentiment-lab.html \
  xx-crypto-lab.html \
  xx-indices-lab.html \
  xx-energy-lab.html \
  xx-precious-metals-lab.html \
  xx-ai-assets-lab.html \
  xx-policy-assets-lab.html

echo
echo "Verification:"
grep -nH "xx-daily-tech-colors-v1\|dailyTechnicalTone\|daily-tech-value" \
  xx-forex-sentiment-lab.html \
  xx-crypto-lab.html \
  xx-indices-lab.html \
  xx-energy-lab.html \
  xx-precious-metals-lab.html \
  xx-ai-assets-lab.html \
  xx-policy-assets-lab.html

echo
echo "No commit or push was performed."
