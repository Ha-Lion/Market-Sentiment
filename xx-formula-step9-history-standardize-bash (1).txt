#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="sentiment-history.html"

if [ ! -f "$FILE" ]; then
  echo "STOP: $FILE not found."
  exit 1
fi

python - <<'PY'
from pathlib import Path

p = Path("sentiment-history.html")
text = p.read_text(encoding="utf-8")

anchor = '''    function stats(points){
      const values = points.map(p=>p.value).filter(Number.isFinite);
'''

helper = '''    function historySentimentGroup(value){
      const label = window.PSDCore && typeof window.PSDCore.classifySentiment === "function"
        ? window.PSDCore.classifySentiment(value)
        : bias(value).text;
      const text = String(label || "");
      if(text.includes("Bullish")) return "bullish";
      if(text.includes("Bearish")) return "bearish";
      return "neutral";
    }

    function historyStatClass(value){
      const group = historySentimentGroup(value);
      return group === "bullish" ? "stat-bull" : (group === "bearish" ? "stat-bear" : "stat-neutral");
    }

'''

if "function historySentimentGroup(value)" not in text:
    if anchor not in text:
        raise SystemExit("STOP: stats anchor not found.")
    text = text.replace(anchor, helper + anchor, 1)

old_stats = '''      const bulls = values.filter(v=>v>=56).length;
      const bears = values.filter(v=>v<45).length;
'''
new_stats = '''      const bulls = values.filter(v=>historySentimentGroup(v)==="bullish").length;
      const bears = values.filter(v=>historySentimentGroup(v)==="bearish").length;
'''
if old_stats not in text:
    raise SystemExit("STOP: stats bull/bear block not found.")
text = text.replace(old_stats, new_stats, 1)

old_tiles = '''      setStatTileClass("statCurrent", s.current>=56 ? "stat-bull" : (s.current<45 ? "stat-bear" : "stat-neutral"));
      setStatTileClass("statChange", s.change>0 ? "stat-bull" : (s.change<0 ? "stat-bear" : "stat-neutral"));
      setStatTileClass("statAverage", s.average>=56 ? "stat-bull" : (s.average<45 ? "stat-bear" : "stat-neutral"));
'''
new_tiles = '''      setStatTileClass("statCurrent", historyStatClass(s.current));
      setStatTileClass("statChange", s.change>0 ? "stat-bull" : (s.change<0 ? "stat-bear" : "stat-neutral"));
      setStatTileClass("statAverage", historyStatClass(s.average));
'''
if old_tiles not in text:
    raise SystemExit("STOP: stat tile classification block not found.")
text = text.replace(old_tiles, new_tiles, 1)

old_bias = '''      const biasItems=[
        {label:"Bullish",value:vals.filter(v=>v>=56).length,color:COLORS.purple},
        {label:"Mixed/Neutral",value:vals.filter(v=>v>=45&&v<56).length,color:COLORS.gold},
        {label:"Bearish",value:vals.filter(v=>v<45).length,color:COLORS.pink}
      ];
'''
new_bias = '''      const biasItems=[
        {label:"Bullish",value:vals.filter(v=>historySentimentGroup(v)==="bullish").length,color:COLORS.purple},
        {label:"Mixed/Neutral",value:vals.filter(v=>historySentimentGroup(v)==="neutral").length,color:COLORS.gold},
        {label:"Bearish",value:vals.filter(v=>historySentimentGroup(v)==="bearish").length,color:COLORS.pink}
      ];
'''
if old_bias not in text:
    raise SystemExit("STOP: bias donut classification block not found.")
text = text.replace(old_bias, new_bias, 1)

old_csv_func = '''    function sentimentLabelForCSV(value){
      if(!Number.isFinite(value)) return "";
      if(window.PSDCore && typeof window.PSDCore.classifySentiment === "function"){
        return window.PSDCore.classifySentiment(value);
      }
      return bias(value).text;
    }
'''
new_csv_func = '''    function sentimentLabelForCSV(value,fallback=""){
      if(!Number.isFinite(value)) return String(fallback || "");
      if(window.PSDCore && typeof window.PSDCore.classifySentiment === "function"){
        return window.PSDCore.classifySentiment(value);
      }
      const local = bias(value).text;
      return local || String(fallback || "");
    }
'''
if old_csv_func not in text:
    raise SystemExit("STOP: CSV sentiment label helper not found.")
text = text.replace(old_csv_func, new_csv_func, 1)

old_global_csv = 'record?.global_sentiment_label || sentimentLabelForCSV(global)'
new_global_csv = 'sentimentLabelForCSV(global, record?.global_sentiment_label || "")'
if old_global_csv not in text:
    raise SystemExit("STOP: global CSV backend-label precedence not found.")
text = text.replace(old_global_csv, new_global_csv, 1)

old_inst_csv = 'item?.sentiment_label || sentimentLabelForCSV(psi)'
new_inst_csv = 'sentimentLabelForCSV(psi, item?.sentiment_label || "")'
if old_inst_csv not in text:
    raise SystemExit("STOP: instrument CSV backend-label precedence not found.")
text = text.replace(old_inst_csv, new_inst_csv, 1)

p.write_text(text, encoding="utf-8")
print("Historical PSI classification standardization prepared.")
PY

git diff --check

echo
echo "Changed file only:"
git status --short -- sentiment-history.html sentiment-core.js

echo
echo "Diff summary:"
git diff --stat -- sentiment-history.html

echo
echo "1) New shared classification helpers:"
grep -nE 'function historySentimentGroup|function historyStatClass|PSDCore.classifySentiment' sentiment-history.html | head -80

echo
echo "2) Remaining local 56/45 classification checks should be gone:"
grep -nE \
  'filter\(v=>v>=56|filter\(v=>v<45|v>=45&&v<56|s\.current>=56|s\.current<45|s\.average>=56|s\.average<45' \
  sentiment-history.html || true

echo
echo "3) Historical stats/bias now use shared grouping:"
grep -nE \
  'historySentimentGroup|historyStatClass|const biasItems=' \
  sentiment-history.html | head -100

echo
echo "4) CSV labels now prefer score-based shared classification:"
grep -nE \
  'function sentimentLabelForCSV|sentimentLabelForCSV\(global, record|sentimentLabelForCSV\(psi, item' \
  sentiment-history.html | head -80

echo
echo "No charts, averages, volatility, momentum, divergence, ribbon/header, CSS, engine data, or sentiment-core.js were modified."
echo "No commit or push was performed."
