#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILES=(
  category-page-core.js
  ai-assets.html
  crypto.html
  energy.html
  forex-sentiment-today.html
  indices.html
  policy-assets.html
  precious-metals.html
)

for f in "${FILES[@]}"; do
  if [ ! -f "$f" ]; then
    echo "STOP: required file missing: $f"
    exit 1
  fi
done

python - <<'PY'
from pathlib import Path

core = Path("category-page-core.js")
text = core.read_text(encoding="utf-8")

old_label = '''function labelForScore(score){
      if(score == null) return "No PSI";
      if(score >= 70) return "Strong Bullish";
      if(score >= 56) return "Bullish";
      if(score >= 45) return "Mixed/Neutral";
      if(score >= 31) return "Bearish";
      return "Strong Bearish";
    }'''

new_label = '''function labelForScore(score,fallback="No PSI"){
      const n=Number(score);
      if(!Number.isFinite(n)) return String(fallback || "No PSI");
      if(global.PSDCore && typeof global.PSDCore.classifySentiment === "function"){
        return global.PSDCore.classifySentiment(n);
      }
      return String(fallback || "No PSI");
    }'''

old_color = '''function colorForScore(score){
      if(score == null) return "#8b949e";
      if(score >= 70) return "#22c55e";
      if(score >= 56) return "#14b8a6";
      if(score >= 45) return "#58a6ff";
      if(score >= 31) return "#f97316";
      return "#ef4444";
    }'''

new_color = '''function colorForScore(score){
      const n=Number(score);
      if(!Number.isFinite(n)) return "#8b949e";
      if(global.PSDCore && typeof global.PSDCore.sentimentColor === "function"){
        return global.PSDCore.sentimentColor(n);
      }
      return "#8b949e";
    }'''

old_bucket = '''function ciBucket(score){
  const n=ciNum(score);
  if(n==null) return "neutral";
  if(n>=56) return "bullish";
  if(n<=44) return "bearish";
  return "neutral";
}'''

new_bucket = '''function ciBucket(score){
  const label=labelForScore(score,"");
  if(label.includes("Bullish")) return "bullish";
  if(label.includes("Bearish")) return "bearish";
  return "neutral";
}'''

for old, new, name in [
    (old_label, new_label, "labelForScore"),
    (old_color, new_color, "colorForScore"),
    (old_bucket, new_bucket, "ciBucket"),
]:
    if old not in text:
        raise SystemExit(f"STOP: expected {name} block not found; no files were written.")
    text = text.replace(old, new, 1)

text = text.replace(
    "/* PublicSentimentDash Category Page Core v5",
    "/* PublicSentimentDash Category Page Core v6",
    1
)

core.write_text(text, encoding="utf-8")

pages = [
    "ai-assets.html",
    "crypto.html",
    "energy.html",
    "forex-sentiment-today.html",
    "indices.html",
    "policy-assets.html",
    "precious-metals.html",
]

old_label_use = 'row.sentiment_label || labelForScore(Number(row.psi))'
new_label_use = 'labelForScore(Number(row.psi), row.sentiment_label || "No PSI")'

for name in pages:
    p = Path(name)
    t = p.read_text(encoding="utf-8")

    count = t.count(old_label_use)
    if count != 2:
        raise SystemExit(
            f"STOP: expected exactly 2 backend-label precedence uses in {name}; found {count}. "
            "category-page-core.js was changed locally, but page files were not written."
        )

    if '<script src="category-page-core.js?v=5"></script>' not in t:
        raise SystemExit(f"STOP: expected category-page-core.js?v=5 script tag not found in {name}.")

    t = t.replace(old_label_use, new_label_use)
    t = t.replace(
        '<script src="category-page-core.js?v=5"></script>',
        '<script src="category-page-core.js?v=6"></script>',
        1
    )
    t = t.replace(
        'category-page-core.js v5 did not load.',
        'category-page-core.js v6 did not load.',
        1
    )
    p.write_text(t, encoding="utf-8")

print("Category PSI standardization prepared.")
PY

git diff --check

echo
echo "Changed files:"
git status --short --   category-page-core.js   ai-assets.html crypto.html energy.html forex-sentiment-today.html   indices.html policy-assets.html precious-metals.html

echo
echo "Diff summary:"
git diff --stat --   category-page-core.js   ai-assets.html crypto.html energy.html forex-sentiment-today.html   indices.html policy-assets.html precious-metals.html

echo
echo "1) Category core now delegates PSI labels/colors to PSDCore:"
grep -nE   'function labelForScore|PSDCore.classifySentiment|function colorForScore|PSDCore.sentimentColor|function ciBucket'   category-page-core.js

echo
echo "2) Duplicate official PSI thresholds should be gone from category-page-core.js:"
grep -nE   'score *>= *70|score *>= *56|score *>= *45|score *>= *31|n *>= *56|n *<= *44'   category-page-core.js || true

echo
echo "3) Category pages now prefer shared score classification, backend label only as fallback:"
grep -nH   'labelForScore(Number(row.psi), row.sentiment_label'   ai-assets.html crypto.html energy.html forex-sentiment-today.html   indices.html policy-assets.html precious-metals.html

echo
echo "4) Cache-busted shared category core:"
grep -nH   'category-page-core.js?v=6'   ai-assets.html crypto.html energy.html forex-sentiment-today.html   indices.html policy-assets.html precious-metals.html

echo
echo "5) Old direct backend-label precedence should be gone:"
grep -nH   'row.sentiment_label || labelForScore(Number(row.psi))'   ai-assets.html crypto.html energy.html forex-sentiment-today.html   indices.html policy-assets.html precious-metals.html || true

echo
echo "No CSS, ribbon/header, engine data, sentiment-core.js, or page layout was modified."
echo "No commit or push was performed."
