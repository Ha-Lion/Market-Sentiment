#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

CORE="sentiment-core.js"
CAT="category-page-core.js"

if [ ! -f "$CORE" ] || [ ! -f "$CAT" ]; then
  echo "STOP: required shared core file missing."
  exit 1
fi

echo "=== CATEGORY PAGE SHARED FORMULA AUDIT — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "1) Production pages loading category-page-core.js:"
for f in ./*.html; do
  base="$(basename "$f")"
  [[ "$base" == xx-* ]] && continue
  if grep -q 'category-page-core.js' "$f"; then
    echo "----- $base -----"
    grep -nE 'sentiment-core.js|category-page-core.js' "$f" || true
  fi
done
echo

echo "2) Local PSI label/color/group functions in category-page-core.js:"
grep -nE \
  'function labelForScore|function colorForScore|function .*tone|function .*sentiment|score *>= *70|score *>= *56|score *>= *45|score *>= *31|n *>= *56|n *<= *44' \
  "$CAT" | head -240 || true
echo

echo "3) Context around the top classification helpers:"
sed -n '1,75p' "$CAT"
echo

echo "4) Context around category PSI average / breadth logic:"
sed -n '510,680p' "$CAT"
echo

echo "5) Direct backend sentiment_label use in category production pages:"
for f in ai-assets.html crypto.html energy.html forex-sentiment-today.html indices.html policy-assets.html precious-metals.html; do
  [ -f "$f" ] || continue
  hits="$(grep -n 'sentiment_label' "$f" || true)"
  if [ -n "$hits" ]; then
    echo "----- $f -----"
    echo "$hits"
  fi
done
echo

echo "6) Calls to labelForScore / colorForScore in category pages and shared core:"
grep -RniE \
  --include='category-page-core.js' \
  --include='ai-assets.html' \
  --include='crypto.html' \
  --include='energy.html' \
  --include='forex-sentiment-today.html' \
  --include='indices.html' \
  --include='policy-assets.html' \
  --include='precious-metals.html' \
  'labelForScore|colorForScore' . | head -360 || true
echo

echo "7) Shared PSDCore functions available:"
grep -nE \
  'const VERSION|function classifySentiment|function badgeClass|function sentimentColor|function countBullBearNeutral|const API' \
  "$CORE" | head -140 || true
echo

echo "8) Check whether category core already references PSDCore:"
grep -nE 'PSDCore|sentiment-core' "$CAT" || true
echo

echo "=== END CATEGORY AUDIT ==="
echo "No files were modified."
