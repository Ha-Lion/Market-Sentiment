#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="xx-sentiment-history-lab.html"

if [ ! -f "$FILE" ]; then
  echo "STOP: $FILE not found."
  exit 1
fi

python - <<'PY'
from pathlib import Path
import re

p = Path("xx-sentiment-history-lab.html")
text = p.read_text(encoding="utf-8")

if 'id="xx-history-final-chart-readability-v2"' in text:
    raise SystemExit("STOP: final chart readability V2 already installed.")

# ---------------------------------------------------------------
# 1) Remove the four duration/subtitle lines if still present.
# ---------------------------------------------------------------
patterns = [
    r'\s*<div class="xx-history-card-sub">Past 7 days[^<]*</div>',
    r'\s*<div class="xx-history-card-sub">Past 3 weeks[^<]*</div>',
    r'\s*<div class="xx-history-card-sub">Past 3 months[^<]*</div>',
    r'\s*<div class="xx-history-card-sub">Past 6 months[^<]*</div>',
]
removed = 0
for pat in patterns:
    text, n = re.subn(pat, '', text, count=1)
    removed += n

# ---------------------------------------------------------------
# 2) Remove older CSS blockers from the glossary parity style.
#    Do not fail if a blocker was already removed.
# ---------------------------------------------------------------
start_marker = '<style id="xx-history-glossary-parity-v1">'
start = text.find(start_marker)
if start == -1:
    raise SystemExit("STOP: glossary parity style block not found.")
end = text.find('</style>', start)
if end == -1:
    raise SystemExit("STOP: glossary parity closing tag not found.")

block = text[start:end]
block = block.replace('transform:none!important', 'transform-origin:center!important')
block = block.replace('box-shadow:none!important', 'box-shadow:none')
text = text[:start] + block + text[end:]

# ---------------------------------------------------------------
# 3) Force the CURRENT canvas textColor assignment to true black
#    in light mode, regardless of its earlier wording.
# ---------------------------------------------------------------
needle = 'const textColor'
pos = text.find(needle)
if pos == -1:
    raise SystemExit("STOP: const textColor assignment not found.")

semi = text.find(';', pos)
if semi == -1:
    raise SystemExit("STOP: textColor assignment semicolon not found.")

line_start = text.rfind('\n', 0, pos) + 1
indent = text[line_start:pos]
current_assignment = text[pos:semi+1]

new_assignment = '''const textColor=isHistoryLight()
        ? "#111111"
        : (getComputedStyle(document.documentElement).getPropertyValue("--hist-muted").trim() || "#d7e0ea");'''

text = text[:pos] + new_assignment + text[semi+1:]

# ---------------------------------------------------------------
# 4) Taller cards, black supporting copy, exact specialty-page
#    glossary motion with no transform/box-shadow blockers.
# ---------------------------------------------------------------
css = r'''
  <style id="xx-history-final-chart-readability-v2">
    #xx-clean-history-dashboard .xx-history-card{
      min-height:330px!important;
    }

    #xx-clean-history-dashboard .xx-chart-wrap{
      height:286px!important;
    }

    #xx-clean-history-dashboard .xx-history-card-head{
      min-height:34px!important;
      padding:8px 10px 3px!important;
      align-items:center!important;
    }

    /* Light mode: make formerly gray supporting text dark/black. */
    html.xx-history-light #xx-clean-history-dashboard .xx-toolbar-label,
    html.xx-history-light #xx-clean-history-dashboard .xx-toolbar-status,
    html.xx-history-light #xx-clean-history-dashboard .xx-chart-legend,
    html.xx-history-light #xx-clean-history-dashboard .xx-legend-item,
    html.xx-history-light #xx-clean-history-dashboard .xx-chart-empty,
    html.xx-history-light .history-lead,
    html.xx-history-light .xx-exact-history summary small,
    html.xx-history-light .xx-exact-help{
      color:#111111!important;
    }

    /*
      Exact specialty-page glossary dance:
      brief 1.11 scale + red glow during a 5-second cycle.
    */
    @keyframes psdPulseGlossary{
      0%,82%,100%{
        transform:scale(1);
        box-shadow:none;
      }
      87%{
        transform:scale(1.11);
        box-shadow:0 0 0 3px rgba(211,47,47,.10),0 0 10px rgba(211,47,47,.32);
      }
      92%{
        transform:scale(1);
        box-shadow:0 0 5px rgba(211,47,47,.18);
      }
    }

    .xx-exact-history .hist-info-btn{
      transform-origin:center!important;
      animation:psdPulseGlossary 5s ease-in-out infinite!important;
    }

    .xx-exact-history .hist-info-btn:hover,
    .xx-exact-history .hist-info-btn:focus-visible,
    .xx-exact-history .hist-info-btn[aria-expanded="true"]{
      animation:none!important;
      transform:scale(1)!important;
    }

    @media(max-width:520px){
      #xx-clean-history-dashboard .xx-history-card{
        min-height:280px!important;
      }
      #xx-clean-history-dashboard .xx-chart-wrap{
        height:236px!important;
      }
    }

    @media(prefers-reduced-motion:reduce){
      .xx-exact-history .hist-info-btn{
        animation:none!important;
      }
    }
  </style>
'''

if '</head>' not in text:
    raise SystemExit("STOP: </head> not found.")
text = text.replace('</head>', css + '\n</head>', 1)

p.write_text(text, encoding="utf-8")

print("Historical Sentiment lab final chart/readability V2 prepared.")
print(f"Duration subtitle lines removed: {removed}")
print("Previous textColor assignment was:")
print(current_assignment.strip())
PY

git diff --check

echo
echo "Changed lab file only:"
git status --short -- xx-sentiment-history-lab.html sentiment-history.html

echo
git diff --stat -- xx-sentiment-history-lab.html

echo
echo "Verification markers:"
grep -n -E \
  'xx-history-final-chart-readability-v2|min-height:330px|height:286px|const textColor=isHistoryLight|animation:psdPulseGlossary 5s' \
  xx-sentiment-history-lab.html | head -80

echo
echo "Duration notes remaining (should print nothing):"
grep -n -E \
  'Past 7 days •|Past 3 weeks •|Past 3 months •|Past 6 months •' \
  xx-sentiment-history-lab.html || true

echo
echo "Production remains untouched."
echo "No commit or push was performed."
