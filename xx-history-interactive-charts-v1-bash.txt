    let xxHistoryModalState=null;

    function xxPointLabel(period,p){
      if(!p) return "";
      if(period==="hourly" && p.ts instanceof Date){
        return new Intl.DateTimeFormat("en-US",{
          month:"short",
          day:"numeric",
          year:"numeric",

    function xxDrawCrosshair(ctx,meta,hover){
      if(!hover || !Number.isInteger(hover.index)) return;
      const p=meta.data[hover.index];
      if(!p) return;

      const {w,h,pad,available,data}=meta;
      const iw=Math.max(1,w-pad.l-pad.r);
      const x=data.length===1
        ? pad.l+iw/2

      const mouseY=Math.max(pad.t,Math.min(h-pad.b,Number(hover.mouseY)||pad.t));

      ctx.save();

      ctx.lineWidth=1;

      ctx.moveTo(x,pad.t);
      ctx.lineTo(x,h-pad.b);

      ctx.beginPath();
      ctx.moveTo(pad.l,mouseY);
      ctx.lineTo(w-pad.r,mouseY);
      ctx.stroke();

      ctx.setLineDash([]);

      const rows=available
        .filter(k=>Number.isFinite(p[k]))
        .map(k=>({
          key:k,
          label:REGION_META[k].label,
          value:Math.round(p[k]),

      const title=xxPointLabel(meta.period,p);
      ctx.font="800 10px Inter,Arial";
      let boxW=ctx.measureText(title).width+22;

      ctx.font="800 10px Inter,Arial";
      rows.forEach(r=>{
        boxW=Math.max(boxW,ctx.measureText(`${r.label}: ${r.value}`).width+34);
      });

      boxW=Math.max(130,Math.min(230,boxW));
      const rowH=16;
      const boxH=25+rows.length*rowH+6;

      let boxX=x+12;
      if(boxX+boxW>w-pad.r) boxX=x-boxW-12;
      boxX=Math.max(pad.l+2,Math.min(w-pad.r-boxW-2,boxX));

      let boxY=mouseY-boxH/2;
      boxY=Math.max(pad.t+2,Math.min(h-pad.b-boxH-2,boxY));

      ctx.fillStyle=isHistoryLight() ? "rgba(255,255,255,.97)" : "rgba(7,13,25,.96)";
      ctx.strokeStyle=isHistoryLight() ? "rgba(51,65,85,.32)" : "rgba(148,163,184,.35)";
      ctx.lineWidth=1;

      ctx.lineTo(boxX+boxW-r,boxY);
      ctx.quadraticCurveTo(boxX+boxW,boxY,boxX+boxW,boxY+r);

    function xxVisibleModalData(){
      if(!xxHistoryModalState) return [];
      return xxHistoryModalState.full.slice(

    function xxModalRangeLabel(data){
      if(!data.length) return "No history";

      return `${first}  →  ${last}  •  ${data.length} points`;
    }

    function xxOpenHistoryModal(period){
      const sets=normalize();
      const full=(sets.full && Array.isArray(sets.full[period]))
        ? sets.full[period]
        : (sets[period] || []);

      if(!full.length) return;

      const current=sets[period] || [];
      const currentCount=Math.max(1,current.length);

      xxHistoryModalState={
        period,
        full,
        start:Math.max(0,full.length-currentCount),
        end:full.length
      };

      const modal=xxEnsureHistoryModal();
      const title=document.getElementById("xxHistoryModalTitle");
      if(title){
        title.textContent=
          period==="hourly" ? "Hourly PSI — Expanded" :
          period==="daily" ? "Daily PSI — Expanded" :
          period==="weekly" ? "Weekly PSI — Expanded" :
          "Monthly PSI — Expanded";
      }

      modal.classList.add("show");
      modal.setAttribute("aria-hidden","false");
      document.body.classList.add("xx-history-modal-open");

      requestAnimationFrame(()=>xxRenderHistoryModal());
    }

    function xxCloseHistoryModal(){
      const modal=document.getElementById("xxHistoryChartModal");
      if(modal){
        modal.classList.remove("show");
        modal.setAttribute("aria-hidden","true");
      }
      document.body.classList.remove("xx-history-modal-open");
      xxHistoryModalState=null;
    }

    function xxRenderHistoryModal(hover=null){
      if(!xxHistoryModalState) return;

      const data=xxVisibleModalData();
      const range=document.getElementById("xxHistoryModalRange");
      if(range) range.textContent=xxModalRangeLabel(data);

      drawChart(
        "xxHistoryModalCanvas",
        "xxHistoryModalEmpty",
        data,
        xxHistoryModalState.period,
        hover
      );
    }

    function xxAdjustModal(action){
      if(!xxHistoryModalState) return;

      const s=xxHistoryModalState;
      const total=s.full.length;
      let count=Math.max(1,s.end-s.start);

      if(action==="all"){
        s.start=0;
        s.end=total;
        xxRenderHistoryModal();
        return;
      }

      if(action==="zoom-in"){
        const next=Math.max(3,Math.round(count*.70));
        s.start=Math.max(0,s.end-next);
      }

      if(action==="zoom-out"){
        const next=Math.min(total,Math.max(count+1,Math.round(count*1.45)));
        const center=(s.start+s.end)/2;
        s.start=Math.max(0,Math.round(center-next/2));
        s.end=Math.min(total,s.start+next);
        s.start=Math.max(0,s.end-next);
      }

      if(action==="earlier" || action==="later"){
        count=Math.max(1,s.end-s.start);
        const step=Math.max(1,Math.round(count*.30));
        const dir=action==="earlier" ? -1 : 1;
        let start=s.start+dir*step;
        start=Math.max(0,Math.min(total-count,start));
        s.start=start;
        s.end=start+count;
      }

      s.start=Math.max(0,Math.min(s.start,total-1));
      s.end=Math.max(s.start+1,Math.min(s.end,total));

      xxRenderHistoryModal();
    }

    function xxBindChartInteraction(canvasId,emptyId,period){
      const canvas=document.getElementById(canvasId);
      if(!canvas || canvas.dataset.xxHistoryInteractive==="1") return;

      canvas.dataset.xxHistoryInteractive="1";

      canvas.addEventListener("mousemove",e=>{
        const meta=xxChartMeta.get(canvasId);
        if(!meta || !meta.data.length) return;

        const rect=canvas.getBoundingClientRect();
        const localX=e.clientX-rect.left;
        const localY=e.clientY-rect.top;
        const iw=Math.max(1,rect.width-meta.pad.l-meta.pad.r);
        const ratio=(localX-meta.pad.l)/iw;
        const index=Math.max(
          0,
          Math.min(
            meta.data.length-1,
            Math.round(ratio*(meta.data.length-1))
          )
        );

        if(canvasId==="xxHistoryModalCanvas"){
          xxRenderHistoryModal({index,mouseY:localY});
        }else{
          drawChart(canvasId,emptyId,meta.data,period,{index,mouseY:localY});
        }
      });

      canvas.addEventListener("mouseleave",()=>{
        const meta=xxChartMeta.get(canvasId);
        if(!meta) return;

        if(canvasId==="xxHistoryModalCanvas"){
          if(xxHistoryModalState) xxRenderHistoryModal();
        }else{
          drawChart(canvasId,emptyId,meta.data,period);
        }
      });

      canvas.addEventListener("click",()=>{
        if(canvasId==="xxHistoryModalCanvas") return;
        xxOpenHistoryModal(period);
      });

      if(canvasId==="xxHistoryModalCanvas"){
        canvas.addEventListener("wheel",e=>{
          if(!xxHistoryModalState) return;
          e.preventDefault();

          const horizontal=Math.abs(e.deltaX)>Math.abs(e.deltaY);
          if(horizontal || e.shiftKey){
            const delta=horizontal ? e.deltaX : e.deltaY;
            xxAdjustModal(delta>0 ? "later" : "earlier");
          }else{
            xxAdjustModal(e.deltaY>0 ? "zoom-out" : "zoom-in");
          }
        },{passive:false});
      }
    }
