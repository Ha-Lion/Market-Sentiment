#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

CORE="sentiment-core.js"
CONFIG="formula-config.json"

if [ ! -f "$CORE" ]; then
  echo "STOP: sentiment-core.js not found."
  exit 1
fi

echo "=== FINAL WEBSITE FORMULA CERTIFICATION SWEEP — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "1) Shared PSI authority:"
grep -nE \
  'const VERSION|function classifySentiment|function calculatePSI|function badgeClass|function sentimentColor|function getTechnicalBias|function buildInstrumentSummary|const API' \
  "$CORE" | head -220 || true
echo

echo "2) Formula config presence / relevant keys:"
if [ -f "$CONFIG" ]; then
  echo "FOUND: $CONFIG"
  grep -nEi \
    'psi|threshold|bull|bear|neutral|mixed|formula|version' \
    "$CONFIG" | head -240 || true
else
  echo "NOT FOUND in website repo: $CONFIG"
fi
echo

echo "3) Production pages/scripts with hard-coded OFFICIAL PSI boundary patterns:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='glossary.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  --exclude='*-lab.js' \
  'score[[:space:]]*(>=|>|<=|<)[[:space:]]*(70|56|45|31|30|44|55|69)|v[[:space:]]*(>=|>|<=|<)[[:space:]]*(70|56|45|31|30|44|55|69)|s[[:space:]]*(>=|>|<=|<)[[:space:]]*(70|56|45|31|30|44|55|69)' \
  . | head -420 || true
echo

echo "4) Static text that explicitly documents PSI ranges:"
grep -RniE \
  --include='*.html' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  '0[[:space:]]*(to|–|-)[[:space:]]*30|31[[:space:]]*(to|–|-)[[:space:]]*44|45[[:space:]]*(to|–|-)[[:space:]]*55|56[[:space:]]*(to|–|-)[[:space:]]*69|70[[:space:]]*(to|–|-)[[:space:]]*100' \
  . | head -320 || true
echo

echo "5) Direct backend sentiment_label precedence outside shared core:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'sentiment_label[[:space:]]*\|\||\|\|[[:space:]]*[A-Za-z0-9_?.]+sentiment_label' \
  . | head -320 || true
echo

echo "6) Local sentiment classifier/helper definitions outside shared core:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'function[[:space:]]+[A-Za-z0-9_]*(sentiment|labelForScore|scoreLabel|classify|bias|bucket)|const[[:space:]]+[A-Za-z0-9_]*(sentiment|labelForScore|scoreLabel|classify|bucket)[[:space:]]*=' \
  . | head -420 || true
echo

echo "7) PSI-derived Technical Direction fallbacks:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'techFromScore|technical.*psi|psi.*technical|score.*return.*Up|score.*return.*Down' \
  . | head -300 || true
echo

echo "8) Technical-derived User Sentiment fallbacks:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'userDefaultFromTech|user.*technical|technical.*user|user.*tech|tech.*user' \
  . | head -320 || true
echo

echo "9) Possible PSI recalculation / aggregation outside shared core:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='sentiment-core.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'headline_psi_score|calculatePSI|weighted.*psi|psi.*weighted|average.*psi|psi.*average|reduce\(.*psi|Math\.round\(.*psi' \
  . | head -420 || true
echo

echo "10) Explicit alternate/secondary model thresholds that may be intentional:"
grep -RniE \
  --include='*.html' \
  --include='*.js' \
  --exclude='xx-*' \
  --exclude='*-lab.html' \
  'Risk-On|Risk-Off|pressureLabel|Impact Score|impactBand|forecast|outlook_score|relevance_score|alignment|momentum|volatility' \
  . | head -420 || true
echo

echo "11) PDF/report formula-related logic:"
if [ -f pdf-report.js ]; then
  grep -nEi \
    'score[[:space:]]*(>=|>|<=|<)|classifySentiment|sentimentColor|sentiment_label|psi|technical|user sentiment' \
    pdf-report.js | head -360 || true
else
  echo "pdf-report.js not present."
fi
echo

echo "12) Production HTML pages using sentiment-core.js:"
for f in ./*.html; do
  base="$(basename "$f")"
  [[ "$base" == xx-* ]] && continue
  [[ "$base" == *-lab.html ]] && continue
  if grep -q 'sentiment-core.js' "$f"; then
    echo "$base"
  fi
done | sort
echo

echo "13) Production HTML pages mentioning PSI/sentiment but NOT loading sentiment-core.js:"
for f in ./*.html; do
  base="$(basename "$f")"
  [[ "$base" == xx-* ]] && continue
  [[ "$base" == *-lab.html ]] && continue
  if grep -qiE 'Public Sentiment Index|\bPSI\b|sentiment' "$f" && ! grep -q 'sentiment-core.js' "$f"; then
    echo "----- $base -----"
    grep -niE 'Public Sentiment Index|\bPSI\b|sentiment' "$f" | head -20 || true
  fi
done
echo

echo "14) Final signal-separation check in Dashboard:"
if [ -f dashboard.html ]; then
  grep -nE \
    'const tech = summary\.tech|const user = actualUser|techFromScore|userDefaultFromTech|score > 58|score < 44' \
    dashboard.html | head -120 || true
fi
echo

echo "=== END FINAL CERTIFICATION SWEEP ==="
echo "No files were modified."
