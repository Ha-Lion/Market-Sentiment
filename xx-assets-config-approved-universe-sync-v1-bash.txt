def norm(v):
    return re.sub(r"[^a-z0-9]+", "", str(v or "").lower())

def values(asset):
    vals = [
        asset.get("id"),

        asset.get("display_name"),

        asset.get("symbol"),

def resolve(category, token):
    key = norm(token)

    # Prefer exact match already in this category.
    local = [a for a in assets if in_category(a, category) and key in values(a)]
    if len(local) == 1:
        return local[0]
    if len(local) > 1:
        raise SystemExit(f"STOP: ambiguous existing {category} match for {token}: {[a.get('id') for a in local]}")

    # Then exact global match.
    global_matches = [a for a in assets if key in values(a)]
    if len(global_matches) == 1:
        return global_matches[0]
    if len(global_matches) > 1:
        # For duplicate symbols, use the candidate whose display/name also equals the token.
        stronger = [
            a for a in global_matches
            if key in {norm(a.get("display_name")), norm(a.get("name")), norm(a.get("data_key"))}
        ]
        if len(stronger) == 1:
            return stronger[0]
        raise SystemExit(f"STOP: ambiguous global match for {token}: {[a.get('id') for a in global_matches]}")

    # Otherwise create only from an explicitly approved definition.
    definition = NEW.get(key)
    if definition is None:
        raise SystemExit(f"STOP: no existing asset and no approved new definition for {category}: {token}")

    # Prevent duplicate id.
    existing_id = next((a for a in assets if a.get("id") == definition["id"]), None)
    if existing_id:
        return existing_id

    row = copy.deepcopy(definition)
    assets.append(row)
    return row
