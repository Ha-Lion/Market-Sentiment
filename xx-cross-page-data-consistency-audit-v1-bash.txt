#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

echo "=== CROSS-PAGE DATA CONSISTENCY AUDIT — READ ONLY ==="
echo
echo "Git:"
git status --short
git log -1 --oneline
echo

echo "1) Shared PSI authority / version"
grep -nE 'const VERSION|function classifySentiment|function calculatePSI|function getTechnicalBias|function buildInstrumentSummary' sentiment-core.js 2>/dev/null || true
echo

echo "2) Production pages loading sentiment-core.js"
for f in index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
         indices.html forex-sentiment-today.html precious-metals.html energy.html crypto.html policy-assets.html ai-assets.html; do
  if [ -f "$f" ]; then
    if grep -q 'sentiment-core.js' "$f"; then
      echo "YES  $f"
    else
      echo "NO   $f"
    fi
  fi
done
echo

echo "3) Live JSON/data sources used by major pages"
grep -nHE 'fetch\([^)]*(dashboard_live|sentiment_history|technical|user|vote|headline|history)[^)]*\.json|dashboard_live\.json|sentiment_history\.json' \
  index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
  watchlist.js category-page-core.js pdf-report.js 2>/dev/null | head -220 || true
echo

echo "4) Local PSI/sentiment classifiers outside shared core"
grep -nHE 'function (labelForScore|scoreLabel|sentimentLabel|bias|historySentimentGroup|ciBucket|sentimentColor)|score *[<>]=? *(30|31|44|45|55|56|69|70)' \
  *.html *.js 2>/dev/null \
  | grep -v 'sentiment-core.js' \
  | head -220 || true
echo

echo "5) Backend sentiment_label / fallback precedence"
grep -nHE 'sentiment_label|global_sentiment_label|europe_sentiment_label|us_sentiment_label' \
  index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
  category-page-core.js pdf-report.js 2>/dev/null | head -220 || true
echo

echo "6) Headline-count logic / sources"
grep -nHE 'headline_count|headlineCount|headlines\.length|top_headlines|relevant.*headline|headline.*count' \
  index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
  category-page-core.js pdf-report.js watchlist.js 2>/dev/null | head -260 || true
echo

echo "7) Technical direction sources / fallbacks"
grep -nHE 'technical_direction|technical.*bias|techBias|getTechnicalBias|technicalMap|Tech |technicalData|technical.*fallback' \
  index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
  category-page-core.js watchlist.js pdf-report.js 2>/dev/null | head -240 || true
echo

echo "8) User sentiment sources / fallbacks"
grep -nHE 'user_sentiment|userSentiment|userMap|vote|voting|fallback.*technical|technical.*fallback' \
  index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
  watchlist.js vote-widget.js pdf-report.js 2>/dev/null | head -240 || true
echo

echo "9) Timestamp / timezone display logic"
grep -nHE 'updated_ny|last.scan|Last Scan|America/New_York|timeZone|toLocaleString|DateTimeFormat|published_at|publishedTime' \
  index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
  category-page-core.js watchlist.js pdf-report.js 2>/dev/null | head -300 || true
echo

echo "10) Instrument names / aliases outside shared core"
grep -nHE 'aliases:\[|aliases: \[|display_name|instrumentAliases|aliasMap|normalizeInstrument|canonical.*instrument' \
  index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html \
  indices.html forex-sentiment-today.html precious-metals.html energy.html crypto.html policy-assets.html ai-assets.html \
  category-page-core.js watchlist.js pdf-report.js 2>/dev/null | head -300 || true
echo

echo "11) PDF/report data sourcing"
grep -nE 'dashboard_live|sentiment_history|PSI_CORE|psi|headline_count|technical|user sentiment|updated_ny|Last Scan' \
  pdf-report.js 2>/dev/null | head -260 || true
echo

echo "12) Major pages: current source-script references"
for f in index.html dashboard.html market-pulse.html market-intelligence.html news-articles.html sentiment-history.html; do
  echo "--- $f ---"
  grep -nE '<script[^>]+src=|fetch\(' "$f" 2>/dev/null | tail -80 || true
done
echo

echo "=== END CROSS-PAGE DATA CONSISTENCY AUDIT ==="
echo "No files were modified."
