table_fn = r'''    function renderExactHistoryTable(){
      const tbody=document.getElementById("xxExactHistoryRows");
      if(!tbody || !historyData) return;

      const rows=(historyData.records || [])
        .filter(r=>r && r.date)
        .slice()
        .sort((a,b)=>String(a.date).localeCompare(String(b.date)));

      if(!rows.length){
        tbody.innerHTML='<tr><td colspan="7">No saved daily PSI records are available.</td></tr>';
        return;
      }

      const value=(r,key)=>{
        const v=Number(r?.[key]);
        return Number.isFinite(v) ? v : null;
      };

      const deltaClass=d=>d>0 ? "xx-delta-up" : d<0 ? "xx-delta-down" : "xx-delta-flat";
      const fmt=v=>Number.isFinite(v) ? String(Math.round(v)) : '<span class="xx-data-missing">—</span>';
      const fmtDelta=d=>{
        if(!Number.isFinite(d)) return '<span class="xx-data-missing">—</span>';
        const prefix=d>0 ? "+" : "";
        return `<span class="${deltaClass(d)}">${prefix}${Math.round(d)}</span>`;
      };

      const rendered=rows.map((r,i)=>{
        const prev=i>0 ? rows[i-1] : null;
        const g=value(r,"global_psi"), pg=value(prev,"global_psi");
        const u=value(r,"us_psi"), pu=value(prev,"us_psi");
        const e=value(r,"europe_psi"), pe=value(prev,"europe_psi");

        const gd=Number.isFinite(g)&&Number.isFinite(pg) ? g-pg : null;
        const ud=Number.isFinite(u)&&Number.isFinite(pu) ? u-pu : null;
        const ed=Number.isFinite(e)&&Number.isFinite(pe) ? e-pe : null;

        const dt=new Date(String(r.date)+"T12:00:00-04:00");
        const dateText=Number.isNaN(dt.getTime())
          ? String(r.date)
          : new Intl.DateTimeFormat("en-US",{month:"short",day:"numeric",year:"numeric"}).format(dt);

        return {
          html:`<tr>
            <td>${dateText}</td>
            <td>${fmt(g)}</td>
            <td>${fmtDelta(gd)}</td>
            <td>${fmt(u)}</td>
            <td>${fmtDelta(ud)}</td>
            <td>${fmt(e)}</td>
            <td>${fmtDelta(ed)}</td>
          </tr>`
        };
      }).reverse();

      tbody.innerHTML=rendered.map(r=>r.html).join("");
    }

'''
