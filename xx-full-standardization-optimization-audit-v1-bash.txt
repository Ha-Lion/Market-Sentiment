def line_hits(text, pattern, regex=False):
    out=[]
    rx=re.compile(pattern, re.I if regex else 0) if regex else None
    for i,line in enumerate(text.splitlines(),1):
        ok = bool(rx.search(line)) if regex else pattern in line
        if ok:
            out.append((i,line.strip()))
    return out

def function_block(text, name):
    # Small extraction, enough to compare local behavior.
    m=re.search(r'function\s+'+re.escape(name)+r'\s*\([^)]*\)\s*\{', text)
    if not m:
        return None
    start=m.start()
    depth=0
    end=None
    in_str=None
    esc=False
    for i,ch in enumerate(text[m.end()-1:], m.end()-1):
        if in_str:
            if esc:
                esc=False
            elif ch=="\\":
                esc=True
            elif ch==in_str:
                in_str=None
            continue
        if ch in ("'",'"','`'):
            in_str=ch
            continue
        if ch=="{":
            depth+=1
        elif ch=="}":
            depth-=1
            if depth==0:
                end=i+1
                break
    return text[start:end] if end else text[start:start+1400]

def norm(v):
