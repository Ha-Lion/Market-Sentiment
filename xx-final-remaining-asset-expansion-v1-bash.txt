import copy
import json
import re
from pathlib import Path

path = Path("assets_config.json")
cfg = json.loads(path.read_text(encoding="utf-8"))
assets = cfg.get("assets")
categories = cfg.get("categories")

if not isinstance(assets, list) or not isinstance(categories, dict):
    raise SystemExit("STOP: unexpected assets_config.json structure.")

def norm(v):
    return re.sub(r"[^a-z0-9]+", "", str(v or "").lower())

def cat_entry(asset, category):
    for c in asset.get("categories", []):
        if isinstance(c, dict) and c.get("category") == category and c.get("enabled", True) is not False:
            return c
    return None

def cat_rows(category):
    rows = []
    for a in assets:
        c = cat_entry(a, category)
        if c and a.get("enabled", True) is not False:
            rows.append((int(c.get("priority", 9999)), a, c))
    rows.sort(key=lambda x: x[0])
    return rows

# Known-good starting point after the already-proven metals batch and added crypto batch.
expected_start = {
    "forex": 15,
    "crypto": 24,
    "indices": 10,
    "energy": 10,
    "precious_metals": 16,
    "ai_assets": 15,
    "policy_geopolitical": 15,
}
for cat, expected in expected_start.items():
    actual = len(cat_rows(cat))
    if actual != expected:
        raise SystemExit(f"STOP: {cat} start count={actual}; expected={expected}")

templates = {cat: copy.deepcopy(cat_rows(cat)[0][2]) for cat in expected_start}

by_id = {a.get("id"): a for a in assets}

def add_category(asset, category, priority):
    # preserve all other memberships
    asset["categories"] = [
        c for c in (asset.get("categories") or [])
        if not (isinstance(c, dict) and c.get("category") == category)
    ]
    c = copy.deepcopy(templates[category])
    c["category"] = category
    c["priority"] = priority
    c["enabled"] = True
    if "role" in c:
        c["role"] = "primary"
    asset["categories"].append(c)
    asset["enabled"] = True

def yahoo(symbol, label, decimals=2, required=False):
    return {
        "source": "yahoo",
        "symbol": symbol,
        "label": label,
        "price_decimals": decimals,
        "data_kind": "market_price",
        "frequency": "daily",
        "retention_days": 180,
        "required": required,
    }

def create_asset(*, aid, name, display, data_key, symbol, atype, aliases,
                 category, priority, market_data=None, notes=""):
    if aid in by_id:
        raise SystemExit(f"STOP: duplicate new id: {aid}")
    row = {
        "id": aid,
        "name": name,
        "display_name": display,
        "data_key": data_key,
        "symbol": symbol,
        "type": atype,
        "enabled": True,
        "aliases": aliases,
        "categories": [],
        "notes": notes,
    }
    if market_data is not None:
        row["market_data"] = market_data
    assets.append(row)
    by_id[aid] = row
    add_category(row, category, priority)
    return row
