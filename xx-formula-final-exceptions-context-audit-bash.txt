#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

echo "=== FINAL FORMULA EXCEPTIONS — EXACT CONTEXT AUDIT — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "1) HOME fallback classification/color blocks:"
if [ -f index.html ]; then
  grep -n -A28 -B6 'function scoreLabel' index.html || true
fi
echo

echo "2) PDF report hard-coded PSI color block:"
if [ -f pdf-report.js ]; then
  grep -n -A22 -B18 'const color = score >= 70' pdf-report.js || true
fi
echo

echo "3) Dashboard Heatwave static range wording + dynamic color authority:"
if [ -f dashboard.html ]; then
  grep -n -A14 -B5 'Sentiment Heatwave' dashboard.html | head -80 || true
  echo
  grep -n -A30 -B3 'function heatColor' dashboard.html || true
fi
echo

echo "4) Public Sentiment Index exact range text + FAQ/schema wording:"
if [ -f public-sentiment-index.html ]; then
  grep -n -A14 -B8 'How to Read PSI' public-sentiment-index.html || true
  echo
  grep -n -A12 -B4 '"What is the Public Sentiment Index?"' public-sentiment-index.html || true
  echo
  grep -n -A10 -B5 'What the Public Sentiment Index Measures' public-sentiment-index.html || true
fi
echo

echo "5) How It Works PSI / User Sentiment wording:"
if [ -f how-it-works.html ]; then
  grep -n -A30 -B8 'Public Sentiment Index' how-it-works.html | head -100 || true
  echo
  grep -n -A18 -B5 'Anonymous User Sentiment' how-it-works.html || true
fi
echo

echo "6) Verify central core already provides classification/color authority:"
if [ -f sentiment-core.js ]; then
  grep -n -A28 -B4 'function classifySentiment' sentiment-core.js || true
  echo
  grep -n -A18 -B3 'function sentimentColor' sentiment-core.js || true
fi
echo

echo "=== END FINAL EXCEPTIONS AUDIT ==="
echo "No files were modified."
