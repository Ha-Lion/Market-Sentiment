#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

SELF="xx-clean-completed-formula-helpers-bash.txt"

FILES=(
  "xx-formula-cleanup-preview-bash.txt"
  "xx-formula-final-centralization-cleanup-bash.txt"
  "xx-formula-final-certification-audit-bash.txt"
  "xx-formula-final-certification-check-bash.txt"
  "xx-formula-final-exceptions-context-audit-bash.txt"
  "xx-formula-sitewide-consistency-audit-bash.txt"
  "xx-formula-step10-dashboard-remaining-audit-bash.txt"
  "xx-formula-step10-dashboard-signal-separation-fix-bash.txt"
  "xx-formula-step10b-dashboard-threshold-usage-audit-bash.txt"
  "xx-formula-step10c-dashboard-exact-context-audit-bash.txt"
  "xx-formula-step10d-dashboard-signal-separation-fix-v2-bash.txt"
  "xx-formula-step4-market-pulse-audit-bash.txt"
  "xx-formula-step4-market-pulse-standardize-bash.txt"
  "xx-formula-step5-market-intelligence-audit-bash.txt"
  "xx-formula-step5-market-intelligence-standardize-bash.txt"
  "xx-formula-step6-watchlist-audit-bash.txt"
  "xx-formula-step6-watchlist-audit-v2-bash.txt"
  "xx-formula-step6-watchlist-final-check-bash.txt"
  "xx-formula-step7-search-world-news-audit-bash.txt"
  "xx-formula-step8-category-core-audit-bash.txt"
  "xx-formula-step8-category-core-standardize-bash.txt"
  "xx-formula-step9-history-remaining-audit-bash.txt"
  "xx-formula-step9-history-standardize-bash (1).txt"
  "xx-formula-step9-home-remaining-audit-bash.txt"
)

PROTECTED=(
  "xx-dashboard-lab.html"
  "xx-dashboard-lab2.html"
  "xx-sentiment-history-lab.html"
  "xx-ribbon-lab.css"
  "xx-site-nav-lab.js"
  "xx-vote-widget-lab.js"
)

echo "=== COMPLETED FORMULA HELPER CLEANUP ==="
echo

if [ -n "$(git status --porcelain)" ]; then
  echo "STOP: repository has uncommitted changes."
  git status --short
  echo "No files were deleted."
  exit 1
fi

echo "Preflight: verifying exact helper list..."
missing=0
for f in "${FILES[@]}"; do
  if ! git ls-files --error-unmatch -- "$f" >/dev/null 2>&1; then
    echo "MISSING/UNTRACKED: $f"
    missing=1
  fi
done

if [ "$missing" -ne 0 ]; then
  echo
  echo "STOP: expected helper list does not match the repository."
  echo "No files were deleted."
  exit 1
fi

echo "Preflight: verifying protected files are not in deletion list..."
for p in "${PROTECTED[@]}"; do
  for f in "${FILES[@]}"; do
    if [ "$p" = "$f" ]; then
      echo "STOP: protected file appeared in deletion list: $p"
      exit 1
    fi
  done
done

echo "Exact files approved for removal: ${#FILES[@]}"
printf '  REMOVE: %s\n' "${FILES[@]}"
echo

git rm -- "${FILES[@]}"

# Remove this cleanup helper too if it was uploaded/tracked.
if git ls-files --error-unmatch -- "$SELF" >/dev/null 2>&1; then
  git rm -- "$SELF"
fi

echo
echo "Protected lab/reference files still present:"
for p in "${PROTECTED[@]}"; do
  if [ -e "$p" ]; then
    echo "  KEEP: $p"
  else
    echo "  WARNING: protected file not present: $p"
  fi
done

echo
echo "Remaining tracked xx-formula-*.txt files:"
remaining="$(git ls-files 'xx-formula-*.txt' || true)"
if [ -n "$remaining" ]; then
  printf '%s\n' "$remaining"
else
  echo "  NONE"
fi

echo
echo "Git status after cleanup:"
git status --short

echo
echo "No production HTML/JS/CSS files were modified."
echo "No commit or push was performed."
