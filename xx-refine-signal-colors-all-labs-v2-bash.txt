MARKET_TONE = '''
    function dailyMarketTone(name){
      const value=String(marketDirection(name)||"Unavailable").trim().toLowerCase();
      if(value.includes("bull") || value.includes("buy") || value.includes("up")) return "market-bullish";
      if(value.includes("bear") || value.includes("sell") || value.includes("down")) return "market-bearish";
      if(value.includes("neutral") || value.includes("mixed") || value.includes("flat")) return "market-neutral";
      return "market-na";
    }

'''

# ---------- preflight ----------
for filename in FILES:
    p=Path(filename)
    if not p.exists():
        raise SystemExit(f"STOP: missing {filename}")

    t=p.read_text(encoding="utf-8")
    if "xx-signal-colors-v2" in t:
        raise SystemExit(f"STOP: {filename} already has V2 signal colors. Nothing changed.")
    if "function dailyTechnicalTone(" not in t:
        raise SystemExit(f"STOP: {filename} missing dailyTechnicalTone(). Nothing changed.")
    if "Daily market insight" not in t or "marketDirection(row.name)" not in t:
        raise SystemExit(f"STOP: {filename} missing Daily Market Insight render row.")
    if "Daily technical direction" not in t or "dailyTechnicalDirection(row.name)" not in t:
        raise SystemExit(f"STOP: {filename} missing Daily Technical Direction render row.")
    if "</head>" not in t:
        raise SystemExit(f"STOP: {filename} missing </head>.")

# ---------- apply ----------
for filename in FILES:
    p=Path(filename)
    t=p.read_text(encoding="utf-8")

    # Put final CSS after the earlier V1 color CSS so V2 wins.
    t=t.replace("</head>", STYLE + "\n</head>", 1)

    # Add Daily Market Insight tone function just before renderTopCards.
    if "function dailyMarketTone(" not in t:
        anchor=re.search(r'\n\s*function\s+renderTopCards\s*\(\)\s*\{',t)
        if not anchor:
            raise SystemExit(f"STOP: renderTopCards() anchor missing in {filename}")
        t=t[:anchor.start()+1] + MARKET_TONE + t[anchor.start()+1:]

    lines=t.splitlines()
    out=[]
    market_hits=0
    tech_hits=0

    for line in lines:
        if 'class="market-insight"' in line and 'marketDirection(row.name)' in line:
            line=line.replace(
                '<div class="market-insight">',
                '<div class="market-insight daily-market-row ${dailyMarketTone(row.name)}">',
                1
            )
            market_hits += 1

        if 'class="market-insight"' in line and 'dailyTechnicalDirection(row.name)' in line:
            line=line.replace(
                '<div class="market-insight">',
                '<div class="market-insight daily-tech-row ${dailyTechnicalTone(row.name)}">',
                1
            )
            line=line.replace(
                '<b class="daily-tech-value ${dailyTechnicalTone(row.name)}">',
                '<b class="daily-tech-value">',
                1
            )
            tech_hits += 1

        out.append(line)

    if market_hits != 1:
        raise SystemExit(f"STOP: {filename} expected 1 market insight row, found {market_hits}")
    if tech_hits != 1:
        raise SystemExit(f"STOP: {filename} expected 1 technical row, found {tech_hits}")

    p.write_text("\n".join(out)+("\n" if t.endswith("\n") else ""),encoding="utf-8")
