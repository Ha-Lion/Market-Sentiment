def norm(v):
    return re.sub(r"[^a-z0-9]+","",str(v or "").lower())

def page_text(path):
    p=Path(path)
    if not p.exists():
        return ""
    return p.read_text(encoding="utf-8",errors="ignore")

def detect_items_var(text):
    m=re.search(r'\b(?:let|const)\s+([A-Z][A-Z0-9_]*_ITEMS)\s*=\s*\[',text)
    return m.group(1) if m else None

def detect_slice_caps(text):
    vals=re.findall(r'\.slice\(\s*0\s*,\s*(\d+)\s*\)',text)
    return vals

def count_local_items(text,varname):
    if not varname:
        return None
    m=re.search(r'\b(?:let|const)\s+'+re.escape(varname)+r'\s*=\s*\[(.*?)\n\s*\];',text,re.S)
    if not m:
        return None
    return len(re.findall(r'\{[^{}]*?name\s*:\s*["\']',m.group(1),re.S))

def has_func(text,name):
    return bool(re.search(r'\bfunction\s+'+re.escape(name)+r'\s*\(',text))

def page_size(text):
    m=re.search(r'aiCardPageSize\s*=\s*window\.innerWidth\s*<=\s*680\s*\?\s*(\d+)\s*:\s*\(window\.innerWidth\s*<=\s*1250\s*\?\s*(\d+)\s*:\s*(\d+)\)',text)
    if m:
        return tuple(map(int,m.groups()))
    m=re.search(r'let\s+aiCardPageSize\s*=\s*(\d+)',text)
    return (None,None,int(m.group(1))) if m else None

print("Current git:")
try:
    print(subprocess.check_output(["git","log","-1","--oneline"], text=True).strip())
except Exception:
    print("unavailable")
print()

cfg_path=Path("assets_config.json")
cfg={}
if cfg_path.exists():
    try:
        cfg=json.loads(cfg_path.read_text(encoding="utf-8"))
    except Exception as e:
        print("WARNING: assets_config.json parse failed:",e)
assets=cfg.get("assets") or []

def category_assets(cat):
    rows=[]
    for a in assets:
        for c in a.get("categories") or []:
            if isinstance(c,dict) and c.get("category")==cat and c.get("enabled",True) is not False and a.get("enabled",True) is not False:
                rows.append((int(c.get("priority",999)),a))
                break
    rows.sort(key=lambda x:x[0])
    return [a for _,a in rows]

def asset_values(a):
    vals=[
        a.get("id"),a.get("name"),a.get("display_name"),a.get("data_key"),a.get("symbol")
    ]
    vals += list(a.get("aliases") or [])
    vals += list(a.get("history_keys") or [])
    md=a.get("market_data") or {}
    vals += [md.get("symbol"),md.get("label")]
    return [norm(v) for v in vals if v]

    print()
    print(f"[{cat}] {page}")

    print("  local item count:",count_local_items(t,var))
    print("  slice caps:",", ".join(detect_slice_caps(t)) or "none")
    print("  page size mobile/tablet/desktop:",page_size(t) or "not detected")
    print("  applyAssetConfig:", "Y" if "applyAssetConfig" in t else "-")

    print("  marketDirection:", "Y" if has_func(t,"marketDirection") else "-")

    print()
    print(f"[{cat}]")

        wanted={norm(display),norm(symbol)}
        present=any(bool(vs & wanted) for vs in value_sets)
        print(f"  {'Y' if present else '-'} {display} ({symbol})")

PY
