#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="dashboard.html"

if [ ! -f "$FILE" ]; then
  echo "STOP: $FILE not found."
  exit 1
fi

echo "=== DASHBOARD THRESHOLD USAGE AUDIT — READ ONLY ==="
echo

echo "Git status:"
git status --short
echo

echo "1) techFromScore definition AND every call:"
grep -nE 'techFromScore\(' "$FILE" || true
echo

echo "2) Context around every techFromScore occurrence:"
while IFS=: read -r line _; do
  [ -n "$line" ] || continue
  start=$(( line > 18 ? line-18 : 1 ))
  end=$(( line+28 ))
  echo "----- around line $line -----"
  sed -n "${start},${end}p" "$FILE"
  echo
done < <(grep -nE 'techFromScore\(' "$FILE" || true)

echo "3) pressureLabel / deepNewsBars / deepTechCompass definitions AND calls:"
grep -nE \
  'pressureLabel\(|deepNewsBars\(|deepTechCompass\(' \
  "$FILE" || true
echo

echo "4) Context around deep-analysis caller(s):"
grep -nE \
  'deepNewsBars\(|deepTechCompass\(|deepAnalysis|renderDeep|openDeep|data\.score' \
  "$FILE" | head -260 || true
echo

echo "5) Instrument score construction / authority:"
grep -nE \
  'buildInstrumentSummary|score:.*summary|score:.*psi|periodScoreFor|data\.score|d\.score' \
  "$FILE" | head -320 || true
echo

echo "6) Heatwave classification functions and their uses:"
grep -nE \
  'heatColor|heatStyle|heat.*label|Sentiment Heatwave|0–30 Bearish|31–44 Mixed Bearish|56–69 Mixed Bullish' \
  "$FILE" | head -260 || true
echo

echo "7) Core PSI display helpers:"
sed -n '1835,1895p' "$FILE"
echo

echo "8) Technical direction source path:"
sed -n '1915,1975p' "$FILE"
echo

echo "9) Deep-analysis threshold context:"
sed -n '2438,2530p' "$FILE"
echo

echo "=== END DASHBOARD USAGE AUDIT ==="
echo "No files were modified."
