#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

PAGE="market-intelligence.html"
CORE="sentiment-core.js"

if [ ! -f "$PAGE" ]; then
  echo "STOP: $PAGE not found."
  echo "Possible intelligence files:"
  find . -maxdepth 1 -type f \( -iname '*intelligence*.html' -o -iname '*market*intel*.html' \) -printf '%f\n' | sort
  exit 1
fi

if [ ! -f "$CORE" ]; then
  echo "STOP: $CORE not found."
  exit 1
fi

echo "=== AI MARKET INTELLIGENCE FORMULA AUDIT — READ ONLY ==="
echo

echo "Repo:"
pwd
echo

echo "Git status:"
git status --short
echo

echo "1) Shared sentiment-core.js / PSDCore usage:"
grep -nE 'sentiment-core|PSDCore|classifySentiment|badgeClass|sentimentColor|buildInstrument|periodScore|headline_psi' "$PAGE" | head -240 || true
echo

echo "2) PSI / sentiment labels and possible thresholds:"
grep -nEi 'strong bullish|bullish|mixed.?neutral|neutral|bearish|strong bearish|sentiment_label|psi.?label|>= *[0-9]+|<= *[0-9]+|> *[0-9]+|< *[0-9]+' "$PAGE" | head -320 || true
echo

echo "3) Possible calculations / derived indicators:"
grep -nEi 'Math\.|average|avg|mean|momentum|volatility|diverg|bias|delta|change|shift|score|psi|weighted|confidence|trend|clamp|normalize|technical|vote' "$PAGE" | head -480 || true
echo

echo "4) Inline functions likely involved in calculations/display:"
grep -nE 'function [A-Za-z0-9_]+\(|const [A-Za-z0-9_]+ *= *\(|=> *\{' "$PAGE" | head -360 || true
echo

echo "5) Data files referenced by AI Market Intelligence:"
grep -oE '[A-Za-z0-9_-]+\.json' "$PAGE" | sort -u
echo

echo "6) Suspicious official PSI boundary numbers:"
grep -nE '(^|[^0-9])(70|69|56|55|45|44|31|30)([^0-9]|$)' "$PAGE" | head -220 || true
echo

echo "7) Any local functions containing label/bias/momentum/score logic:"
grep -nEi 'function .*label|function .*bias|function .*momentum|function .*score|const .*label.*=>|const .*bias.*=>|const .*momentum.*=>|const .*score.*=>' "$PAGE" | head -260 || true
echo

echo "8) Core formula version / available shared functions:"
grep -nE 'const VERSION|function classifySentiment|function badgeClass|function sentimentColor|function buildInstrumentSeries|function buildInstrumentPeriodScore|function buildAuditSnapshot|const API' "$CORE" | head -160 || true
echo

echo "=== END AUDIT ==="
echo "No files were modified."
