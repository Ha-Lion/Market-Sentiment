#!/usr/bin/env bash
set -euo pipefail

if [ "$(basename "$PWD")" != "Market-Sentiment" ]; then
  echo "STOP: run this from /workspaces/Market-Sentiment"
  exit 1
fi

FILE="sentiment-history.html"

python - <<'PY'
from pathlib import Path

p = Path("sentiment-history.html")
text = p.read_text(encoding="utf-8")

old_title = '<span>Global vs U.S. flow</span>'
new_title = '<span>Global vs. U.S. PSI — Last 4 Full Weeks</span>'

if old_title not in text:
    raise SystemExit("STOP: expected Chart 2 title not found.")
text = text.replace(old_title, new_title, 1)

anchor = '    function periodLabel(period){\n'
if anchor not in text:
    raise SystemExit("STOP: periodLabel anchor not found.")

if "function aggregateWeeklyComparison(" not in text:
    fn = '''    function aggregateWeeklyComparison(records){
      const groups = new Map();

      records
        .filter(r =>
          r &&
          r.ts instanceof Date &&
          !Number.isNaN(r.ts.getTime()) &&
          Number.isFinite(r.global) &&
          Number.isFinite(r.us)
        )
        .forEach(r => {
          const key = isoWeekKey(r.ts);
          if(!groups.has(key)){
            groups.set(key,{
              key,
              label:key.replace("-W"," W"),
              global:[],
              us:[],
              dates:new Set(),
              lastTs:r.ts
            });
          }

          const g = groups.get(key);
          g.global.push(r.global);
          g.us.push(r.us);
          g.dates.add(r.date);

          if(r.ts > g.lastTs) g.lastTs = r.ts;
        });

      return [...groups.values()]
        .filter(g => g.dates.size >= 7)
        .map(g => ({
          key:g.key,
          label:g.label,
          sort:g.key,
          count:g.dates.size,
          lastTs:g.lastTs,
          global:avg(g.global),
          us:avg(g.us)
        }))
        .sort((a,b) =>
          String(a.sort).localeCompare(String(b.sort)) ||
          a.lastTs-b.lastTs
        );
    }

'''
    text = text.replace(anchor, fn + anchor, 1)

old_draw = '''      const weeklyGlobal=aggregate("global","weekly",true).slice(-4);
      const weeklyUs=aggregate("us","weekly",true).slice(-4);
      drawLine(document.getElementById("chartFlow"),[
        {points:weeklyGlobal,color:COLORS.gold,fill:"rgba(255,215,128,.16)",width:2.2},
        {points:weeklyUs,color:COLORS.pink,fill:"rgba(255,63,134,.11)",width:2.2}
      ],{dots:false,scan:true,pad:{l:34,r:10,t:30,b:20}});
'''

new_draw = '''      const weeklyComparison=aggregateWeeklyComparison(filterByRange(state.raw)).slice(-4);
      const weeklyGlobal=weeklyComparison.map(p=>({...p,value:p.global}));
      const weeklyUs=weeklyComparison.map(p=>({...p,value:p.us}));
      drawLine(document.getElementById("chartFlow"),[
        {points:weeklyGlobal,color:COLORS.gold,fill:"rgba(255,215,128,.16)",width:2.2},
        {points:weeklyUs,color:COLORS.pink,fill:"rgba(255,63,134,.11)",width:2.2}
      ],{dots:false,scan:true,labelAll:true,pad:{l:34,r:10,t:30,b:20}});
'''

if old_draw not in text:
    raise SystemExit("STOP: expected Chart 2 drawing block not found.")

text = text.replace(old_draw, new_draw, 1)
p.write_text(text, encoding="utf-8")
print("Chart 2 weekly comparison prepared.")
PY

git diff --check

echo
echo "Changed file only:"
git status --short
echo
git diff --stat -- "$FILE"
echo
echo "Chart 2 markers:"
grep -n -E 'Last 4 Full Weeks|aggregateWeeklyComparison|weeklyComparison=' "$FILE"
echo
echo "No commit or push was performed."
